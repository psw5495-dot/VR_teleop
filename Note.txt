document.getElementById('btn-vr').addEventListener('click', async () => {
    log('🎯 VR 버튼 클릭 감지됨!', 'ok');

    if (!navigator.xr) {
        log('❌ navigator.xr 없음 - WebXR 미지원', 'err');
        return;
    }

    try {
        // 지원 여부 재확인
        const supported = await navigator.xr.isSessionSupported('immersive-vr');
        log(`immersive-vr 지원: ${supported}`, supported ? 'ok' : 'err');

        if (!supported) {
            log('💡 immersive-vr 미지원 - Guardian 설정 확인 필요', 'warn');
            return;
        }

        log('VR 세션 시작 시도 중...', 'info');

        // 세션 시작 (호환성 향상을 위해 minimal options)
        const session = await navigator.xr.requestSession('immersive-vr', {
            optionalFeatures: ['local-floor']
        });

        log('✅ VR 세션 생성 성공!', 'ok');
        await renderer.xr.setSession(session);
        log('✅ 렌더러 연결 완료 - VR 모드 진입!', 'ok');

        // UI 숨기기
        document.querySelector('h1').style.display = 'none';
        document.getElementById('log-container').style.display = 'none';
        statusDiv.style.display = 'none';
        document.querySelectorAll('.btn').forEach(btn => btn.style.display = 'none');
        renderer.domElement.style.display = 'block';

        // 세션 종료 이벤트
        session.addEventListener('end', () => {
            log('VR 세션 종료됨', 'warn');
            // UI 복원
            document.querySelector('h1').style.display = 'block';
            document.getElementById('log-container').style.display = 'block';
            statusDiv.style.display = 'block';
            document.querySelectorAll('.btn').forEach(btn => btn.style.display = 'block');
            renderer.domElement.style.display = 'none';
            statusDiv.textContent = 'VR 세션 종료됨';
            statusDiv.style.background = 'rgba(255, 152, 0, 0.3)';
        });

    } catch (error) {
        log(`❌ VR 세션 실패: ${error.name}`, 'err');
        log(`상세 메시지: ${error.message}`, 'err');

        const solutions = {
            'NotAllowedError': '🔧 Guardian 설정 또는 VR 권한 허용 필요',
            'NotSupportedError': '🔧 개발자 모드 활성화 필요',
            'SecurityError': '🔧 HTTPS 주소 확인 필요',
            'InvalidStateError': '🔧 Quest 브라우저 완전 재시작 필요'
        };

        const solution = solutions[error.name] || '🔧 Quest 재부팅 후 재시도';
        log(solution, 'warn');

        statusDiv.textContent = `❌ VR 시작 실패: ${error.name}`;
        statusDiv.style.background = 'rgba(244, 67, 54, 0.4)';
    }
});
