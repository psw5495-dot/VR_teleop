#!/usr/bin/env python3
# ~/TeleVision/g1_bridge_node.py
"""
Meta Quest3 → G1 Teleoperation 브릿지 (Unitree 공식 예제 구조 적용)
핵심: 500Hz 제어 루프 + 올바른 관절 인덱스 + 자세 유지
"""

import sys
import time
import numpy as np
import threading
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState

# ── Unitree SDK2 (공식 예제와 동일한 임포트) ─────────────────
from unitree_sdk2py.core.channel import (
    ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
)
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC
from unitree_sdk2py.utils.thread import RecurrentThread
from unitree_sdk2py.comm.motion_switcher.motion_switcher_client import MotionSwitcherClient

# ── G1 로봇 사양 (공식 예제와 동일) ──────────────────────────
G1_NUM_MOTOR = 29

# 공식 예제의 Kp/Kd 값
Kp = [
    60, 60, 60, 100, 40, 40,      # left leg
    60, 60, 60, 100, 40, 40,      # right leg  
    60, 40, 40,                   # waist
    40, 40, 40, 40, 40, 40, 40,   # left arm (15~21)
    40, 40, 40, 40, 40, 40, 40    # right arm (22~28) ⭐️
]

Kd = [
    1, 1, 1, 2, 1, 1,     # left leg
    1, 1, 1, 2, 1, 1,     # right leg
    1, 1, 1,              # waist
    1, 1, 1, 1, 1, 1, 1,  # left arm
    1, 1, 1, 1, 1, 1, 1   # right arm ⭐️
]

class Mode:
    PR = 0  # Series Control
    AB = 1  # Parallel Control

# ⭐️ 올바른 G1 관절 인덱스 (공식 예제 확인됨)
class G1JointIndex:
    # 왼팔 (기존에 잘못 오른팔로 사용하던 인덱스)
    LeftShoulderPitch = 15
    LeftShoulderRoll = 16
    LeftShoulderYaw = 17
    LeftElbow = 18
    LeftWristRoll = 19
    LeftWristPitch = 20
    LeftWristYaw = 21
    
    # 진짜 오른팔 ⭐️
    RightShoulderPitch = 22
    RightShoulderRoll = 23
    RightShoulderYaw = 24
    RightElbow = 25
    RightWristRoll = 26
    RightWristPitch = 27
    RightWristYaw = 28

# ROS2 관절 이름 → G1 모터 인덱스 매핑 (수정됨)
ROS_TO_G1_INDEX = {
    'right_shoulder_pitch_joint': G1JointIndex.RightShoulderPitch,  # 22
    'right_shoulder_roll_joint' : G1JointIndex.RightShoulderRoll,   # 23
    'right_shoulder_yaw_joint'  : G1JointIndex.RightShoulderYaw,    # 24
    'right_elbow_pitch_joint'   : G1JointIndex.RightElbow,          # 25
    'right_wrist_roll_joint'    : G1JointIndex.RightWristRoll,      # 26
    'right_wrist_pitch_joint'   : G1JointIndex.RightWristPitch,     # 27
    'right_wrist_yaw_joint'     : G1JointIndex.RightWristYaw,       # 28
}

# 왼팔 홈 포지션 (자연스러운 자세)
LEFT_ARM_HOME = {
    G1JointIndex.LeftShoulderPitch: 0.0,
    G1JointIndex.LeftShoulderRoll:  0.3,   # 약간 벌린 자세
    G1JointIndex.LeftShoulderYaw:   0.0,
    G1JointIndex.LeftElbow:         0.5,   # 약간 구부린 자세
    G1JointIndex.LeftWristRoll:     0.0,
    G1JointIndex.LeftWristPitch:    0.0,
    G1JointIndex.LeftWristYaw:      0.0,
}


class G1TeleopBridge(Node):
    def __init__(self):
        super().__init__('g1_teleop_bridge')
        
        # ── Unitree 예제와 동일한 제어 변수 ─────────────────────
        self.control_dt_ = 0.002       # 500Hz 제어 주기
        self.init_duration_ = 3.0      # 초기화 시간 (홈 포지션으로 이동)
        self.time_ = 0.0
        
        self.mode_pr_ = Mode.PR
        self.mode_machine_ = 0
        self.update_mode_machine_ = False
        
        # Unitree 메시지 객체
        self.low_cmd = LowCmd_()
        self.low_state = None
        self.crc = CRC()
        
        # ── 텔레오퍼레이션 상태 ────────────────────────────────
        self.target_q = [0.0] * G1_NUM_MOTOR  # 목표 관절 각도
        self.initial_posture_captured = False
        self.teleop_active = False
        self.emergency_stop = False
        
        # 통계
        self.cmd_count = 0
        self.last_log = time.time()
        
        # ROS2 구독 (IK 노드에서 계산된 관절 각도 수신)
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', 
            self.ros_joint_callback, 10
        )
        
        self.get_logger().info("🤖 G1 Teleop Bridge 초기화 완료")
        self.get_logger().info("⚠️  로봇 주변 안전 확인 후 시작하세요!")

    def Init(self):
        """Unitree 예제와 동일한 초기화 구조"""
        
        # ── MotionSwitcherClient로 High-level 모드 해제 ────────
        self.get_logger().info("🔍 G1 모션 제어 권한 획득 중...")
        self.msc = MotionSwitcherClient()
        self.msc.SetTimeout(5.0)
        self.msc.Init()

        max_retry = 3
        for attempt in range(max_retry):
            try:
                status, result = self.msc.CheckMode()
                self.get_logger().info(f"   CheckMode 시도 {attempt+1}: status={status}")
                
                if result is None:
                    self.get_logger().warn(f"   응답 없음 ({attempt+1}/{max_retry})")
                    if attempt < max_retry - 1:
                        time.sleep(2)
                        continue
                    else:
                        self.get_logger().warn("   MotionSwitcher 건너뜀")
                        break
                
                # 현재 모션이 있으면 해제
                if result.get('name', ''):
                    self.get_logger().info(f"   현재 모션 '{result['name']}' 해제 중...")
                    self.msc.ReleaseMode()
                    time.sleep(1)
                else:
                    self.get_logger().info("   ✅ 이미 Low-level 모드")
                    break
                    
            except Exception as e:
                self.get_logger().warn(f"   CheckMode 예외: {e}")
                if attempt == max_retry - 1:
                    self.get_logger().warn("   기본값으로 진행")
                break

        # ── DDS 퍼블리셔/서브스크라이버 생성 ─────────────────────
        self.lowcmd_publisher_ = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.lowcmd_publisher_.Init()

        self.lowstate_subscriber = ChannelSubscriber("rt/lowstate", LowState_)
        self.lowstate_subscriber.Init(self.LowStateHandler, 10)
        
        self.get_logger().info("✅ DDS 채널 초기화 완료")

    def Start(self):
        """Unitree 예제와 동일한 시작 구조"""
        
        # 500Hz 제어 스레드 생성
        self.lowCmdWriteThreadPtr = RecurrentThread(
            interval=self.control_dt_,
            target=self.LowCmdWrite,
            name="g1_control"
        )
        
        # G1 상태 수신 대기
        self.get_logger().info("⏳ G1 로봇 상태 수신 대기 중...")
        timeout = 10.0
        elapsed = 0.0
        
        while not self.update_mode_machine_:
            time.sleep(0.5)
            elapsed += 0.5
            
            if elapsed % 2 == 0:
                self.get_logger().info(f"   대기 중... ({elapsed:.0f}s/{timeout:.0f}s)")
                
            if elapsed >= timeout:
                self.get_logger().warn("   타임아웃 - 기본값으로 진행")
                self.mode_machine_ = 0
                self.update_mode_machine_ = True
                break

        self.get_logger().info(f"✅ mode_machine: {self.mode_machine_}")
        self.get_logger().info("🚀 500Hz 제어 루프 시작!")
        
        # 키보드 비상 정지 모니터링
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        
        self.lowCmdWriteThreadPtr.Start()

    def LowStateHandler(self, msg: LowState_):
        """G1 로봇 현재 상태 수신 (500Hz)"""
        self.low_state = msg

        # 최초 수신 시 mode_machine 동기화 및 현재 자세 캡처
        if not self.update_mode_machine_:
            self.mode_machine_ = self.low_state.mode_machine
            
            # ⭐️ 로봇의 현재 관절 각도를 목표로 초기화 (갑작스러운 움직임 방지)
            for i in range(G1_NUM_MOTOR):
                self.target_q[i] = self.low_state.motor_state[i].q
                
            self.initial_posture_captured = True
            self.update_mode_machine_ = True

    def ros_joint_callback(self, msg):
        """ROS2에서 VR IK 목표 각도 수신 (약 30Hz)"""
        if not self.initial_posture_captured or self.emergency_stop:
            return
            
        matched_joints = 0
        
        # ⭐️ 수신된 오른팔 관절 각도를 target_q에 업데이트
        for joint_name, g1_idx in ROS_TO_G1_INDEX.items():
            if joint_name in msg.name:
                idx_in_msg = msg.name.index(joint_name)
                
                # 안전 범위 체크 (±π 라디안)
                target_angle = float(msg.position[idx_in_msg])
                if abs(target_angle) > np.pi:
                    self.get_logger().warn(f"⚠️ {joint_name}: {target_angle:.3f}rad 범위 초과")
                    continue
                
                self.target_q[g1_idx] = target_angle
                matched_joints += 1
        
        # 통계 로그
        now = time.time()
        if now - self.last_log > 3.0:
            self.get_logger().info(f"📡 VR 명령 수신: {matched_joints}개 관절 업데이트")
            self.last_log = now

    def LowCmdWrite(self):
        """
        500Hz 메인 제어 루프 (Unitree 예제 구조 유지)
        Stage 1: 홈 포지션으로 이동
        Stage 2: VR 텔레오퍼레이션
        """
        if not self.initial_posture_captured or self.low_state is None:
            return

        self.time_ += self.control_dt_

        if self.emergency_stop:
            self._send_hold_position()
            return

        if self.time_ < self.init_duration_:
            # ── Stage 1: 홈 포지션으로 부드럽게 이동 ──────────────
            self._stage_init_home()
        else:
            # ── Stage 2: VR 텔레오퍼레이션 ─────────────────────────
            if not self.teleop_active:
                self.teleop_active = True
                self.get_logger().info("🎮 VR 텔레오퍼레이션 모드 활성화!")
                self.get_logger().info("   Quest3 컨트롤러를 움직여 G1 오른팔을 제어하세요")
            
            self._stage_teleop()

        # ⭐️ CRC 계산 후 DDS 전송 (예제와 동일)
        self.low_cmd.crc = self.crc.Crc(self.low_cmd)
        self.lowcmd_publisher_.Write(self.low_cmd)
        self.cmd_count += 1

    def _stage_init_home(self):
        """Stage 1: 안전한 홈 포지션으로 이동"""
        ratio = np.clip(self.time_ / self.init_duration_, 0.0, 1.0)

        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        # 모든 관절 기본 설정
        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]
            
            # 현재 위치에서 시작
            current_q = self.low_state.motor_state[i].q
            self.low_cmd.motor_cmd[i].q = current_q

        # ⭐️ 왼팔만 홈 포지션으로 부드럽게 이동
        for motor_idx, home_q in LEFT_ARM_HOME.items():
            current_q = self.low_state.motor_state[motor_idx].q
            self.low_cmd.motor_cmd[motor_idx].q = (
                (1.0 - ratio) * current_q + ratio * home_q
            )
            # target_q도 업데이트 (Stage 2에서 사용)
            self.target_q[motor_idx] = self.low_cmd.motor_cmd[motor_idx].q

    def _stage_teleop(self):
        """Stage 2: VR 컨트롤러로 오른팔 제어"""
        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        # ⭐️ 모든 관절을 target_q 각도로 설정
        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].q = self.target_q[i]  # VR 명령 또는 홈 포지션
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]

    def _send_hold_position(self):
        """비상 정지: 현재 위치 유지"""
        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].q = self.low_state.motor_state[i].q
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]

    def keyboard_monitor(self):
        """키보드 비상 정지 모니터링"""
        print("\n⌨️  키보드 제어:")
        print("   s + Enter: 비상 정지")
        print("   r + Enter: 제어 재개")
        print("   q + Enter: 종료\n")
        
        try:
            while True:
                key = input().strip().lower()
                if key in ['s', 'stop']:
                    self.emergency_stop = True
                    self.get_logger().warn('🛑 비상 정지 활성화!')
                elif key in ['r', 'resume']:
                    self.emergency_stop = False
                    self.get_logger().info('▶️ 제어 재개')
                elif key in ['q', 'quit']:
                    self.get_logger().info('🛑 종료 중...')
                    exit(0)
        except (EOFError, KeyboardInterrupt):
            pass


def main():
    print("=" * 70)
    print("🤖 Meta Quest3 → G1 Teleoperation 브릿지 시작")
    print("⚠️  주의: 로봇 주변에 장애물이 없는지 확인하세요!")
    print("=" * 70)
    
    # ── 1. 네트워크 인터페이스 지정 ──────────────────────────────
    if len(sys.argv) > 1:
        iface = sys.argv[1]
        print(f"🌐 지정된 네트워크 인터페이스: {iface}")
    else:
        print("❌ 네트워크 인터페이스를 인자로 전달해주세요.")
        print("   사용법: python3 g1_bridge_node.py enx80691ad9f138")
        print("   (성공했던 유선 인터페이스 이름을 사용하세요)")
        sys.exit(1)

    # DDS 초기화 (예제와 동일)
    ChannelFactoryInitialize(0, iface)
    time.sleep(0.5)

    # ── 2. ROS2 노드 및 G1 제어 초기화 ──────────────────────────
    rclpy.init()
    bridge_node = G1TeleopBridge()
    
    try:
        bridge_node.Init()
        bridge_node.Start()
        
        # ROS2 콜백 스핀 (VR 데이터 수신)
        rclpy.spin(bridge_node)
        
    except KeyboardInterrupt:
        print("\n🛑 브릿지 노드 종료 중...")
    finally:
        bridge_node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
