#!/usr/bin/env python3
# ~/g1_teleop/g1_ik_node.py
"""
G1 역기구학 노드 - 모든 들여쓰기 버그 수정 완료
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from sensor_msgs.msg import JointState
import numpy as np
import pinocchio as pin
import os
import time
import threading
import xml.etree.ElementTree as ET


class G1IKNode(Node):
    def __init__(self):
        super().__init__('g1_ik_node')
        
        self.setup_pinocchio_model()
        
        self.target_sub = self.create_subscription(
            PoseStamped, '/g1/target_pose', self.target_callback, 10)
        self.joint_pub = self.create_publisher(
            JointState, '/g1/arm/joint_commands', 10)
        self.joint_state_sub = self.create_subscription(
            JointState, '/joint_states', self.joint_state_callback, 10)
        
        self.emergency_stop = False
        self.robot_connected = False
        self.last_log_time = time.time()
        self.prev_q = None
        
        self.workspace_limits = {
            'x_min': 0.05, 'x_max': 1.0,
            'y_min': -0.8, 'y_max': 0.8,
            'z_min': 0.1,  'z_max': 1.5
        }
        
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        
        self.get_logger().info('🤖 G1 IK 노드 시작!')
        self.get_logger().info('📡 /g1/target_pose 구독 중...')
        self.get_logger().info('⚠️  s=비상정지 | r=재개')

    def setup_pinocchio_model(self):
        """URDF 로드 및 모델 초기화"""
        base_dir = os.path.dirname(os.path.abspath(__file__))
        candidates = [
            os.path.join(base_dir, 'models', 'g1', 'g1_body29_hand14.urdf'),
            os.path.join(base_dir, 'models', 'g1', 'g1_body23.urdf'),
        ]
        
        urdf_path = None
        for path in candidates:
            if os.path.exists(path):
                urdf_path = path
                break
        
        if not urdf_path:
            self.get_logger().error('❌ URDF 파일 없음!')
            self.get_logger().error('G1에서 복사: scp -r unitree@[G1_IP]:'
                                  '/home/unitree/xr_teleoperate/assets/g1 '
                                  './models/')
            exit(1)
        
        try:
            self.model = pin.buildModelFromUrdf(urdf_path)
            self.data = self.model.createData()
            self.current_q = pin.neutral(self.model)
            
            self.get_logger().info(f'✅ URDF 로드: {os.path.basename(urdf_path)}')
            self.get_logger().info(f'   관절 수: {self.model.nq}')
        except Exception as e:
            self.get_logger().error(f'❌ URDF 파싱 실패: {e}')
            exit(1)
        
        self.right_arm_joints = self._parse_right_arm_joints(urdf_path)
        self.ee_frame_id = self._find_ee_frame()

    def _parse_right_arm_joints(self, urdf_path):
        """오른팔 관절 파싱"""
        target_joints = [
            'right_shoulder_pitch_joint', 'right_shoulder_roll_joint',
            'right_shoulder_yaw_joint',   'right_elbow_pitch_joint',
            'right_wrist_roll_joint',     'right_wrist_pitch_joint',
            'right_wrist_yaw_joint'
        ]
        
        try:
            tree = ET.parse(urdf_path)
            root = tree.getroot()
            
            valid_joints = []
            for target_name in target_joints:
                for joint in root.findall('joint'):
                    if (joint.get('name') == target_name and
                        joint.get('type') in ['revolute', 'continuous']):
                        valid_joints.append(target_name)
                        break
            
            self.get_logger().info(f'✅ 오른팔 관절 {len(valid_joints)}개: {valid_joints}')
            return valid_joints
            
        except Exception as e:
            self.get_logger().error(f'관절 파싱 에러: {e}')
            return []

    def _find_ee_frame(self):
        """End-effector 프레임 탐색"""
        candidates = [
            'right_wrist_yaw_link',   'right_wrist_pitch_link',
            'right_wrist_roll_link',  'right_hand_base_link',
            'right_hand_link',        'right_elbow_pitch_link'
        ]
        
        for name in candidates:
            if self.model.existFrame(name):
                frame_id = self.model.getFrameId(name)
                self.get_logger().info(f'✅ EE 프레임: {name} (ID: {frame_id})')
                return frame_id
        
        # 디버깅용 프레임 목록
        self.get_logger().warn('⚠️ 표준 EE 프레임 없음. 사용 가능한 프레임:')
        for i in range(min(15, self.model.nframes)):
            self.get_logger().info(f'   [{i}] {self.model.frames[i].name}')
        
        frame_id = self.model.nframes - 1
        self.get_logger().warn(f'⚠️ 마지막 프레임 사용: {self.model.frames[frame_id].name}')
        return frame_id

    def keyboard_monitor(self):
        """비상 정지 키보드 모니터링"""
        try:
            while rclpy.ok():
                key = input().strip().lower()
                if key in ['s', 'stop']:
                    self.emergency_stop = True
                    self.get_logger().warn('🛑 비상 정지!')
                elif key in ['r', 'start']:
                    self.emergency_stop = False
                    self.get_logger().info('▶️ 재개')
        except Exception:
            pass

    def joint_state_callback(self, msg):
        """관절 상태 업데이트"""
        if not self.robot_connected:
            self.robot_connected = True
            self.get_logger().info('✅ G1 관절 상태 수신!')

    def check_workspace_limits(self, pos):
        """작업 공간 안전 제한"""
        x, y, z = pos
        lim = self.workspace_limits
        
        if not (lim['x_min'] <= x <= lim['x_max']):
            return False, f"x={x:.3f} 범위 초과"
        if not (lim['y_min'] <= y <= lim['y_max']):
            return False, f"y={y:.3f} 범위 초과"
        if not (lim['z_min'] <= z <= lim['z_max']):
            return False, f"z={z:.3f} 범위 초과"
        
        return True, "OK"

    def solve_ik(self, target_pos):
        """Pinocchio IK 계산"""
        q = self.current_q.copy()
        eps, max_iter, dt, damp = 1e-3, 50, 0.1, 1e-4
        
        for _ in range(max_iter):
            pin.forwardKinematics(self.model, self.data, q)
            pin.updateFramePlacements(self.model, self.data)
            
            current_pos = self.data.oMf[self.ee_frame_id].translation
            error = target_pos - current_pos
            
            if np.linalg.norm(error) < eps:
                return q, True
            
            J = pin.computeFrameJacobian(
                self.model, self.data, q, self.ee_frame_id,
                pin.ReferenceFrame.LOCAL_WORLD_ALIGNED
            )[:3, :]
            
            dq = J.T @ np.linalg.solve(J @ J.T + damp * np.eye(3), error)
            q = np.clip(q + dt * dq,
                       self.model.lowerPositionLimit,
                       self.model.upperPositionLimit)
        
        return q, False

    def target_callback(self, msg):
        """목표 좌표 → 관절 각도 변환"""
        if self.emergency_stop:
            return
        
        try:
            target_pos = np.array([
                msg.pose.position.x,
                msg.pose.position.y,
                msg.pose.position.z
            ])
            
            # 안전 제한 확인
            ok, reason = self.check_workspace_limits(target_pos)
            if not ok:
                self.get_logger().warn(f'⚠️ 안전 제한: {reason}')
                return
            
            # IK 계산
            q_target, converged = self.solve_ik(target_pos)
            if q_target is None:
                return
            
            # 급격한 각도 변화 방지
            MAX_DELTA = 0.1  # rad/step
            if self.prev_q is not None:
                diff = q_target - self.prev_q
                max_change = np.max(np.abs(diff))
                if max_change > MAX_DELTA:
                    q_target = self.prev_q + (diff / max_change) * MAX_DELTA
            
            self.publish_joint_command(q_target)
            self.current_q = q_target
            self.prev_q = q_target.copy()
            
            # 주기적 로그
            now = time.time()
            if now - self.last_log_time > 2.0:
                status = "수렴" if converged else "근사"
                self.get_logger().info(
                    f'✅ IK {status} → pos=({target_pos[0]:.3f}, '
                    f'{target_pos[1]:.3f}, {target_pos[2]:.3f})'
                )
                self.last_log_time = now
                
        except Exception as e:
            self.get_logger().error(f'target_callback 에러: {e}')

    def publish_joint_command(self, q):
        """관절 명령 발행"""
        if not self.right_arm_joints:
            return
        
        msg = JointState()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.name = self.right_arm_joints
        n = len(self.right_arm_joints)
        msg.position = q[:n].tolist()
        msg.velocity = [0.0] * n
        
        self.joint_pub.publish(msg)


def main():
    rclpy.init()
    node = G1IKNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        node.get_logger().info('🛑 IK 노드 종료')
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
