class G1PosePublisher(Node):
    def __init__(self):
        super().__init__('g1_pose_publisher')
        
        # G1 IK 노드가 구독할 목표 좌표 토픽
        self.target_pub = self.create_publisher(
            PoseStamped, '/g1/target_pose', 10
        )
        
        # 원시 VR 데이터도 발행 (디버깅용)
        self.vr_pub = self.create_publisher(
            PoseStamped, '/vr/raw_pose', 10
        )
        
        self.msg_count = 0
        self.get_logger().info('✅ G1 Pose Publisher 시작')
        self.get_logger().info('📡 토픽: /g1/target_pose, /vr/raw_pose')

    def publish_poses(self, result):
        """teleop_pipeline 결과를 ROS 2 메시지로 발행"""
        if not result:
            return
            
        current_time = self.get_clock().now().to_msg()
        
        # 1. 필터링된 목표 좌표 발행 (IK 노드용)
        if 'target_pose' in result:
            target = result['target_pose']
            msg = PoseStamped()
            msg.header.stamp = current_time
            msg.header.frame_id = 'g1_base_link'
            
            msg.pose.position.x = float(target['x'])
            msg.pose.position.y = float(target['y'])
            msg.pose.position.z = float(target['z'])
            msg.pose.orientation.x = float(target.get('qx', 0))
            msg.pose.orientation.y = float(target.get('qy', 0))
            msg.pose.orientation.z = float(target.get('qz', 0))
            msg.pose.orientation.w = float(target.get('qw', 1))
            
            self.target_pub.publish(msg)
        
        # 2. 원시 VR 데이터 발행 (디버깅용)
        if 'raw_vr_pose' in result:
            raw = result['raw_vr_pose']
            vr_msg = PoseStamped()
            vr_msg.header.stamp = current_time
            vr_msg.header.frame_id = 'quest3_world'
            
            vr_msg.pose.position.x = float(raw['x'])
            vr_msg.pose.position.y = float(raw['y'])
            vr_msg.pose.position.z = float(raw['z'])
            vr_msg.pose.orientation.x = float(raw.get('qx', 0))
            vr_msg.pose.orientation.y = float(raw.get('qy', 0))
            vr_msg.pose.orientation.z = float(raw.get('qz', 0))
            vr_msg.pose.orientation.w = float(raw.get('qw', 1))
            
            self.vr_pub.publish(vr_msg)
        
        self.msg_count += 1
        if self.msg_count % 150 == 0:  # 5초마다 통계
            self.get_logger().info(f'📊 {self.msg_count}개 메시지 발행')

# 전역 ROS 노드
ros_publisher = None

def ros_spin_thread():
    """ROS 2 노드를 별도 스레드에서 실행"""
    global ros_publisher
    rclpy.init()
    ros_publisher = G1PosePublisher()
    try:
        rclpy.spin(ros_publisher)
    except KeyboardInterrupt:
        pass
    finally:
        ros_publisher.destroy_node()
        rclpy.shutdown()



class Handler(SimpleHTTPRequestHandler):

    def do_GET(self):
        if self.path in ('/', '/index.html'):
            html = self._get_webxr_html()
            encoded = html.encode('utf-8')
            self.send_response(200)
            self.send_header('Content-Type', 'text/html; charset=utf-8')
            self.send_header('Content-Length', str(len(encoded)))
            self.end_headers()
            self.wfile.write(encoded)
        elif self.path == '/status':
            status = {"server": "running", "pipeline": "ready", "time": time.time()}
            response = json.dumps(status, indent=2).encode('utf-8')
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Content-Length', str(len(response)))
            self.end_headers()
            self.wfile.write(response)
        else:
            self.send_response(404)
            self.end_headers()

    def do_POST(self):
        if self.path == "/pose":
            length = int(self.headers.get("Content-Length", "0"))
            raw = self.rfile.read(length)
            try:
                data = json.loads(raw.decode("utf-8"))
                frame = parse_frame_from_json(data)
                result = pipeline.process(frame)

                # ⭐️ ROS 2로 좌표 발행
                if ROS2_AVAILABLE and ros_publisher:
                    ros_publisher.publish_poses(result)

                print("\n=== TELEOP RESULT ===")
                print(json.dumps(result, indent=2, ensure_ascii=False))

                response = json.dumps(result).encode("utf-8")
                self.send_response(200)
                self.send_header("Content-Type", "application/json")
                self.send_header("Content-Length", str(len(response)))
                self.send_header("Access-Control-Allow-Origin", "*")
                self.end_headers()
                self.wfile.write(response)

            except Exception as e:
                print("POST parse/process error:", e)
                err = json.dumps({"ok": False, "error": str(e)}).encode("utf-8")
                self.send_response(500)
                self.send_header("Content-Type", "application/json")
                self.send_header("Content-Length", str(len(err)))
                self.end_headers()
                self.wfile.write(err)

        elif self.path == "/reset":
            pipeline.reset_origin()
            response = json.dumps({"ok": True, "message": "Origin reset"}).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(response)))
            self.end_headers()
            self.wfile.write(response)

        else:
            self.send_response(404)
            self.end_headers()

    def log_message(self, format, *args):
        if args and 'POST' in str(args[0]):
            super().log_message(format, *args)

function log(msg) {
    const d = document.getElementById('log');

function setStatus(msg, color) {

// 1단계: 환경 검사 (AR/VR 모두 확인)

async function runCheck() {
    log('=== 환경 검사 시작 ===');

    const isHttps = window.location.protocol === 'https:';
    log((isHttps ? '✅' : '❌') + ' HTTPS: ' + window.location.protocol);
    if (!isHttps) { setStatus('❌ HTTPS 필수!', 'rgba(244,67,54,0.4)'); return; }

    if (!navigator.xr) {
        log('❌ navigator.xr 없음 (WebXR 미지원)');
        setStatus('❌ WebXR 미지원 브라우저', 'rgba(244,67,54,0.4)');
        return;
    }
    log('✅ navigator.xr 확인됨');

    // ⭐️ AR(패스스루) 지원 확인
    try {
        const arSupported = await navigator.xr.isSessionSupported('immersive-ar');
        log((arSupported ? '✅' : '⚠️') + ' immersive-ar (패스스루): ' + 
            (arSupported ? '지원됨 → MR 모드 사용 가능!' : '미지원 → VR 모드만 가능'));
    } catch(e) {
        log('⚠️ AR 지원 확인 실패: ' + e.message);
    }

    // VR 지원 확인 (fallback용)
    try {
        const vrSupported = await navigator.xr.isSessionSupported('immersive-vr');
        log((vrSupported ? '✅' : '❌') + ' immersive-vr: ' + 
            (vrSupported ? '지원됨' : '미지원'));

        if (vrSupported) {
            setStatus('✅ 모든 검사 통과 - MR 또는 VR 시작 가능!', 'rgba(76,175,80,0.4)');
            log('🎉 2단계 MR 시작을 먼저 시도해보세요!');
        } else {
            setStatus('❌ VR/AR 미지원 - Guardian 설정 확인', 'rgba(244,67,54,0.4)');
        }
    } catch(e) {
        log('❌ 지원 확인 실패: ' + e.message);
    }
}

// ============================================================
// ⭐️ 2단계: MR 시작 (패스스루 - 현실 배경)
// ============================================================
async function startAR() {
    log('🌍 MR(패스스루) 세션 시작 시도...');
    if (!navigator.xr) { log('❌ navigator.xr 없음'); return; }

    try {
        // ⭐️ immersive-ar 세션 요청 (패스스루 모드)
        xrSession = await navigator.xr.requestSession('immersive-ar', {
            optionalFeatures: ['local-floor']
        });
        log('✅ MR 세션 생성 성공!');

        await setupXRSession(true);  // true = AR 모드

    } catch(e) {
        log('❌ MR 시작 실패: ' + e.name + ' - ' + e.message);

        // ⭐️ AR 실패 시 자동으로 VR 모드로 전환 (사용자 편의성)
        if (e.name === 'NotSupportedError') {
            log('⚠️ 패스스루 미지원 → VR 모드로 자동 전환...');
            setTimeout(() => startVR(), 1000);
        } else {
            const solutions = {
                'NotAllowedError' : 'Guardian 설정 또는 패스스루 권한 허용 필요',
                'SecurityError'   : 'HTTPS 주소 확인 필요',
                'InvalidStateError': '브라우저 완전 재시작 필요'
            };
            log('💡 ' + (solutions[e.name] || 'Quest 재부팅 후 재시도'));
            setStatus('❌ MR 시작 실패: ' + e.name, 'rgba(244,67,54,0.4)');
        }
    }
}

            'NotSupportedError': '개발자 모드 활성화 필요',
            'SecurityError'   : 'HTTPS 주소 확인 필요',
            'InvalidStateError': '브라우저 완전 재시작 필요'

        log('💡 ' + (solutions[e.name] || 'Quest 재부팅 후 재시도'));

    }
}

// ⭐️ XR 세션 공통 설정 함수 (AR/VR 통합)
// ============================================================
async function setupXRSession(isAR) {
    // 1) WebGL 컨텍스트 생성 (무한 로딩 해결 핵심)
    const canvas = document.createElement('canvas');
    gl = canvas.getContext('webgl', { xrCompatible: true });
    if (!gl) throw new Error('WebGL 컨텍스트 생성 실패');

    // 2) XRWebGLLayer 생성 및 연결
    const layer = new XRWebGLLayer(xrSession, gl);
    await xrSession.updateRenderState({ baseLayer: layer });
    log('✅ WebGL 렌더링 레이어 연결 완료! (모드: ' + (isAR ? 'AR' : 'VR') + ')');

    // 3) 참조 공간 설정
    try {
        xrRefSpace = await xrSession.requestReferenceSpace('local-floor');
        log('✅ local-floor 참조 공간 설정');
    } catch(_) {
        xrRefSpace = await xrSession.requestReferenceSpace('local');
        log('✅ local 참조 공간 설정 (fallback)');
    }

    // 4) UI 전환
    ['title','status','btn-check','btn-ar','btn-vr','btn-reset','log']
        .forEach(id => {
            const el = document.getElementById(id);
            if (el) el.style.display = 'none';
        });

    if (isAR) {
        document.body.style.background = 'transparent';
    } else {
        document.body.style.background = '#111';
    }

    // 5) 세션 종료 이벤트
    xrSession.addEventListener('end', () => {
        log((isAR ? 'MR' : 'VR') + ' 세션 종료됨');
        location.reload();
    });

    // 6) 렌더링 루프 시작
    xrSession.requestAnimationFrame((time, frame) => onXRFrame(time, frame, isAR));
    log('🚀 ' + (isAR ? 'MR(현실 배경)' : 'VR(검은 배경)') + ' 컨트롤러 추적 시작!');
}

// ============================================================
// XR 프레임 루프 (AR/VR 통합)
// ============================================================
function onXRFrame(time, frame, isAR) {
    if (!xrSession || !gl) return;
    xrSession.requestAnimationFrame((t, f) => onXRFrame(t, f, isAR));

    // ⭐️ 화면 그리기 (무한 로딩 방지 + AR/VR 구분)
    gl.bindFramebuffer(gl.FRAMEBUFFER,
        xrSession.renderState.baseLayer.framebuffer);

    if (isAR) {
        // ⭐️ AR 모드: 완전 투명 (현실 배경이 그대로 보임)
        gl.clearColor(0.0, 0.0, 0.0, 0.0);
    } else {
        // VR 모드: 어두운 회색
        gl.clearColor(0.05, 0.05, 0.05, 1.0);
    }
    gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

    // 30Hz 전송 제한
    if (time - lastSendTime < 1000 / SEND_HZ) return;
    lastSendTime = time;

    // 오른손 컨트롤러 데이터 추출
    for (const src of xrSession.inputSources) {
        if (src.handedness !== 'right' || !src.gripSpace) continue;

        const pose = frame.getPose(src.gripSpace, xrRefSpace);
        if (!pose) continue;

        const p = pose.transform.position;
        const o = pose.transform.orientation;
        sendPose('right', p.x, p.y, p.z, o.x, o.y, o.z, o.w);

        frameCount++;
        if (frameCount % 150 === 0) {
            console.log('📊 ' + (isAR ? 'MR' : 'VR') + ' 데이터 전송 중:', {
                x: p.x.toFixed(3), y: p.y.toFixed(3), z: p.z.toFixed(3)
            });
        }
        break;
    }
}

// ============================================================
// 서버로 포즈 전송 (기존과 동일)
// ============================================================
function sendPose(hand, x, y, z, qx, qy, qz, qw) {
    fetch('/pose', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            hand, x, y, z, qx, qy, qz, qw,
            timestamp: Date.now() / 1000
        })
    })
    .then(r => r.json())
    .then(result => {
        if (result.status === 'watchdog_home')
            console.warn('⚠️ 워치독 발동 - 홈 위치로 복귀');
    })
    .catch(e => console.debug('전송 에러:', e.message));
}
