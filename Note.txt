#!/usr/bin/env python3
# ~/TeleVision/g1_bridge_node.py
"""
Meta Quest3 → G1 Teleoperation (공식 예제 패턴 적용)
핵심: 안정적인 LowCmd 생성 + 전체 관절 안전 제어
"""

import time
import sys
import numpy as np
import threading
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState

# ⭐️ 공식 예제와 동일한 임포트 패턴
from unitree_sdk2py.core.channel import (
    ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
)
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_
from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowState_
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_
from unitree_sdk2py.utils.crc import CRC
from unitree_sdk2py.utils.thread import RecurrentThread
from unitree_sdk2py.comm.motion_switcher.motion_switcher_client import MotionSwitcherClient

# ── G1 로봇 사양 (공식 예제와 동일) ──────────────────────────
G1_NUM_MOTOR = 29

Kp = [
    60, 60, 60, 100, 40, 40,      # left leg
    60, 60, 60, 100, 40, 40,      # right leg
    60, 40, 40,                   # waist
    40, 40, 40, 40, 40, 40, 40,   # left arm
    40, 40, 40, 40, 40, 40, 40    # right arm
]

Kd = [
    1, 1, 1, 2, 1, 1,     # left leg
    1, 1, 1, 2, 1, 1,     # right leg
    1, 1, 1,              # waist
    1, 1, 1, 1, 1, 1, 1,  # left arm
    1, 1, 1, 1, 1, 1, 1   # right arm
]

class Mode:
    PR = 0

# 오른팔 관절 인덱스 (22~28번)
RIGHT_ARM_INDICES = [22, 23, 24, 25, 26, 27, 28]

ROS_TO_G1_INDEX = {
    'right_shoulder_pitch_joint': 22,
    'right_shoulder_roll_joint' : 23,
    'right_shoulder_yaw_joint'  : 24,
    'right_elbow_pitch_joint'   : 25,
    'right_wrist_roll_joint'    : 26,
    'right_wrist_pitch_joint'   : 27,
    'right_wrist_yaw_joint'     : 28,
}


class G1TeleopBridge(Node):
    def __init__(self):
        super().__init__('g1_teleop_bridge')
        
        # ⭐️ 공식 예제와 동일한 제어 변수
        self.time_ = 0.0
        self.control_dt_ = 0.002  # 500Hz
        self.duration_ = 3.0      # 워밍업 시간
        
        self.mode_machine_ = 0
        self.update_mode_machine_ = False
        
        # ⭐️ 핵심: 공식 예제와 동일한 객체 생성 방식
        self.low_cmd = unitree_hg_msg_dds__LowCmd_()
        self.low_state = None
        self.crc = CRC()
        
        # 관절 각도 상태
        self.initial_q = [0.0] * G1_NUM_MOTOR
        self.target_q = [0.0] * G1_NUM_MOTOR
        self.current_smooth_q = [0.0] * G1_NUM_MOTOR
        
        # 증분 제어 파라미터 (부드러운 움직임)
        self.MAX_DELTA = 0.002  # 기존 0.0005 → 0.002 (4배 빠르게)
        self.teleop_active = False
        self.emergency_stop = False
        
        # 통계 및 디버깅
        self.cmd_count = 0
        self.last_log = time.time()
        self.vr_data_count = 0
        
        # ROS2 구독
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', 
            self.ros_joint_callback, 10
        )
        
        self.get_logger().info("🤖 G1 Teleop Bridge 초기화 완료")
        self.get_logger().info("⭐ 공식 예제 패턴 적용 - 안정성 향상")

    def Init(self):
        """⭐️ 공식 예제와 동일한 초기화 패턴"""
        self.msc = MotionSwitcherClient()
        self.msc.SetTimeout(5.0)
        self.msc.Init()

        # 공식 예제와 동일한 모드 해제 방식
        status, result = self.msc.CheckMode()
        while result and result.get('name', ''):
            self.get_logger().info(f"모션 '{result['name']}' 해제 중...")
            self.msc.ReleaseMode()
            status, result = self.msc.CheckMode()
            time.sleep(1)

        # 공식 예제와 동일한 채널 생성
        self.lowcmd_publisher_ = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.lowcmd_publisher_.Init()

        self.lowstate_subscriber = ChannelSubscriber("rt/lowstate", LowState_)
        self.lowstate_subscriber.Init(self.LowStateHandler, 10)
        
        self.get_logger().info("✅ DDS 채널 초기화 완료")

    def Start(self):
        """⭐️ 공식 예제와 동일한 시작 구조"""
        self.lowCmdWriteThreadPtr = RecurrentThread(
            interval=self.control_dt_,
            target=self.LowCmdWrite,
            name="control"
        )
        
        # 공식 예제와 동일한 대기 방식
        while not self.update_mode_machine_:
            time.sleep(1)

        self.get_logger().info("🚀 500Hz 제어 루프 시작!")
        self.get_logger().info("📋 VR 명령 대기 중 (로봇은 현재 자세 유지)")
        
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        self.lowCmdWriteThreadPtr.Start()

    def LowStateHandler(self, msg: LowState_):
        """공식 예제와 동일한 상태 핸들러"""
        self.low_state = msg

        if not self.update_mode_machine_:
            self.mode_machine_ = msg.mode_machine
            
            # 현재 로봇 자세를 모든 배열에 동일하게 설정
            for i in range(G1_NUM_MOTOR):
                q = msg.motor_state[i].q
                self.initial_q[i] = q
                self.target_q[i] = q
                self.current_smooth_q[i] = q
            
            self.update_mode_machine_ = True
            self.get_logger().info("✅ 현재 자세로 초기화 완료 (움직임 없음)")

    def ros_joint_callback(self, msg):
        """VR에서 오른팔 목표 각도 수신 - 디버깅 강화"""
        if not self.update_mode_machine_:
            return
            
        self.vr_data_count += 1
        matched_joints = 0
        significant_changes = {}
        
        # 오른팔 관절의 target_q만 VR 값으로 업데이트
        for joint_name, g1_idx in ROS_TO_G1_INDEX.items():
            if joint_name in msg.name:
                idx = msg.name.index(joint_name)
                new_angle = float(msg.position[idx])
                old_angle = self.target_q[g1_idx]
                
                # 안전 범위 확인 (±π 라디안)
                if abs(new_angle) > np.pi:
                    self.get_logger().warn(f"⚠️ {joint_name}: {new_angle:.3f}rad 범위 초과")
                    continue
                
                # 의미 있는 변화량 추적
                diff = abs(new_angle - old_angle)
                if diff > 0.001:  # 0.001 라디안 이상 변화
                    short_name = joint_name.replace('right_', '').replace('_joint', '')
                    significant_changes[short_name] = f"{old_angle:.4f}→{new_angle:.4f}"
                
                self.target_q[g1_idx] = new_angle
                matched_joints += 1
        
        # ⭐️ 상세한 실시간 로그
        now = time.time()
        if now - self.last_log > 1.0:  # 1초마다
            if significant_changes:
                self.get_logger().info(
                    f"📡 VR 수신 #{self.vr_data_count} | "
                    f"관절: {matched_joints}개 | 🎯 변화: {significant_changes}"
                )
                if not self.teleop_active:
                    self.teleop_active = True
                    self.get_logger().info("🎮 VR 텔레오퍼레이션 시작!")
            else:
                self.get_logger().warn(
                    f"📡 VR 수신 #{self.vr_data_count} | "
                    f"관절: {matched_joints}개 | ❌ 의미 있는 변화 없음"
                )
            self.last_log = now

    def LowCmdWrite(self):
        """⭐️ 공식 예제 구조 + 개선된 텔레오퍼레이션"""
        if not self.update_mode_machine_ or self.low_state is None:
            return
            
        self.time_ += self.control_dt_

        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        if self.emergency_stop:
            # 비상 정지: 현재 smooth 각도 유지
            for i in range(G1_NUM_MOTOR):
                self.low_cmd.motor_cmd[i].mode = 1
                self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                self.low_cmd.motor_cmd[i].dq = 0.0
                self.low_cmd.motor_cmd[i].tau = 0.0
                self.low_cmd.motor_cmd[i].kp = Kp[i]
                self.low_cmd.motor_cmd[i].kd = Kd[i]
        elif self.time_ < self.duration_:
            # Stage 1: 워밍업 (공식 예제와 동일 - 안정성을 위해 필요)
            ratio = np.clip(self.time_ / self.duration_, 0.0, 1.0)
            
            for i in range(G1_NUM_MOTOR):
                self.low_cmd.motor_cmd[i].mode = 1
                self.low_cmd.motor_cmd[i].tau = 0.0
                self.low_cmd.motor_cmd[i].dq = 0.0
                self.low_cmd.motor_cmd[i].kp = Kp[i]
                self.low_cmd.motor_cmd[i].kd = Kd[i]
                
                # 현재 위치에서 초기 자세로 부드럽게 이동
                current_pos = self.low_state.motor_state[i].q
                self.low_cmd.motor_cmd[i].q = (
                    (1.0 - ratio) * current_pos + ratio * self.initial_q[i]
                )
                self.current_smooth_q[i] = self.low_cmd.motor_cmd[i].q
        else:
            # Stage 2: VR 텔레오퍼레이션
            for i in range(G1_NUM_MOTOR):
                # ⭐️ 모든 관절 활성화 (G1 안전 메커니즘 요구사항)
                self.low_cmd.motor_cmd[i].mode = 1
                self.low_cmd.motor_cmd[i].dq = 0.0
                self.low_cmd.motor_cmd[i].tau = 0.0
                self.low_cmd.motor_cmd[i].kp = Kp[i]
                self.low_cmd.motor_cmd[i].kd = Kd[i]

                if i in RIGHT_ARM_INDICES:
                    # 오른팔: VR 목표를 향해 증분 이동
                    target = self.target_q[i]
                    current = self.current_smooth_q[i]
                    diff = target - current
                    
                    # ⭐️ 증분 제어 (부드러운 움직임)
                    if abs(diff) > self.MAX_DELTA:
                        self.current_smooth_q[i] = (
                            current + np.sign(diff) * self.MAX_DELTA
                        )
                    else:
                        self.current_smooth_q[i] = target
                    
                    self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                else:
                    # 나머지: 초기 자세 유지 (안전)
                    self.low_cmd.motor_cmd[i].q = self.initial_q[i]

        # ⭐️ 공식 예제와 동일한 CRC 계산 + 전송
        self.low_cmd.crc = self.crc.Crc(self.low_cmd)
        self.lowcmd_publisher_.Write(self.low_cmd)
        self.cmd_count += 1

        # 주기적 상태 로그
        if self.cmd_count % 2500 == 0:  # 5초마다
            # 오른팔 현재 각도 표시
            arm_angles = []
            for i in RIGHT_ARM_INDICES:
                arm_angles.append(f"{self.current_smooth_q[i]:.3f}")
            
            self.get_logger().info(
                f"✅ DDS 전송 {self.cmd_count}회 | "
                f"CRC: 0x{self.low_cmd.crc:08X} | "
                f"오른팔: [{', '.join(arm_angles)}]"
            )

    def keyboard_monitor(self):
        """실시간 파라미터 조절"""
        print("\n⌨️  키보드 제어:")
        print("   + : 속도 증가 (x1.5)")
        print("   - : 속도 감소 (÷1.5)")
        print("   s : 비상 정지")
        print("   r : 제어 재개")
        print(f"   현재 MAX_DELTA: {self.MAX_DELTA:.5f} rad/tick\n")
        
        try:
            while True:
                key = input().strip().lower()
                if key == '+':
                    self.MAX_DELTA = min(self.MAX_DELTA * 1.5, 0.02)
                    self.get_logger().info(f'⚡ 속도 증가: {self.MAX_DELTA:.5f}')
                elif key == '-':
                    self.MAX_DELTA = max(self.MAX_DELTA / 1.5, 0.0001)
                    self.get_logger().info(f'🐢 속도 감소: {self.MAX_DELTA:.5f}')
                elif key == 's':
                    self.emergency_stop = True
                    self.get_logger().warn('🛑 비상 정지!')
                elif key == 'r':
                    self.emergency_stop = False
                    self.get_logger().info('▶️ 제어 재개')
        except (EOFError, KeyboardInterrupt):
            pass


def main():
    print("=" * 70)
    print("🤖 Meta Quest3 → G1 Teleoperation (공식 예제 패턴)")
    print("⚠️  주의: 로봇 주변에 장애물이 없는지 확인하세요!")
    print("=" * 70)
    
    if len(sys.argv) > 1:
        ChannelFactoryInitialize(0, sys.argv[1])
    else:
        ChannelFactoryInitialize(0)

    rclpy.init()
    node = G1TeleopBridge()
    
    try:
        node.Init()
        node.Start()
        rclpy.spin(node)
    except KeyboardInterrupt:
        print("\n🛑 브릿지 노드 종료 중...")
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
