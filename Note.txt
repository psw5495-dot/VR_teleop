#!/usr/bin/env python3
"""
teleop_pipeline.py
Quest 3 VR 데이터를 로봇 제어 좌표로 변환하는 완전한 파이프라인
- 손떨림 필터링 (EMA)
- 안전 제한 (최대 이동 거리)
- 좌표계 변환 (VR ↔ 로봇)
- 워치독 (연결 끊김 감지)
"""

import time
from dataclasses import dataclass
from typing import Tuple, Dict, Any, Optional

# ==========================================
# 설정 및 데이터 구조
# ==========================================

@dataclass
class TeleopConfig:
    """텔레오퍼레이션 파이프라인 설정"""
    
    # 스케일링 (VR 움직임 → 로봇 움직임 비율)
    pos_scale: float = 0.5
    
    # 로봇 홈 포지션 (기준점)
    robot_home_pos: Tuple[float, float, float] = (0.35, -0.20, 0.45)
    
    # 안전 제한 (한 프레임당 최대 이동 거리, 단위: m)
    max_dx: float = 0.10
    max_dy: float = 0.10
    max_dz: float = 0.10
    
    # EMA 필터 계수 (0~1: 작을수록 부드럽지만 지연 증가)
    ema_alpha: float = 0.15
    
    # 워치독 타임아웃 (초): 이 시간 동안 데이터 없으면 홈으로 복귀
    watchdog_timeout_s: float = 0.3
    
    # 좌표축 부호 변환 (VR ↔ 로봇 좌표계 맞추기)
    axis_sign: Tuple[float, float, float] = (1.0, -1.0, -1.0)


@dataclass
class TeleopFrame:
    """Quest 3에서 수신한 VR 프레임 데이터"""
    x: float = 0.0
    y: float = 0.0
    z: float = 0.0
    qx: float = 0.0
    qy: float = 0.0
    qz: float = 0.0
    qw: float = 1.0
    hand: str = 'right'
    timestamp: float = 0.0
    tracked: bool = True


# ==========================================
# JSON 파싱 함수
# ==========================================

def parse_frame_from_json(data: Dict[str, Any]) -> TeleopFrame:
    """
    Quest 3에서 전송된 JSON 데이터를 TeleopFrame으로 변환
    
    지원 형식:
    1. Flat 형식: {"hand": "right", "x": 0.1, "y": 0.2, "z": 0.3, ...}
    2. Nested 형식: {"right": {"position": {"x": 0.1, ...}, ...}}
    """
    
    frame = TeleopFrame()
    frame.timestamp = data.get('timestamp', time.time())
    
    # ----- Flat 형식 파싱 (사용자 코드에서 사용하는 형식) -----
    if 'hand' in data or 'x' in data:
        frame.x = float(data.get('x', 0.0))
        frame.y = float(data.get('y', 0.0))
        frame.z = float(data.get('z', 0.0))
        frame.qx = float(data.get('qx', 0.0))
        frame.qy = float(data.get('qy', 0.0))
        frame.qz = float(data.get('qz', 0.0))
        frame.qw = float(data.get('qw', 1.0))
        frame.hand = data.get('hand', 'right')
        frame.tracked = True
        return frame
    
    # ----- Nested 형식 파싱 -----
    if 'right' in data:
        right = data['right']
        if 'position' in right:
            pos = right['position']
            frame.x = float(pos.get('x', 0.0))
            frame.y = float(pos.get('y', 0.0))
            frame.z = float(pos.get('z', 0.0))
        
        if 'quaternion' in right:
            quat = right['quaternion']
            frame.qx = float(quat.get('x', 0.0))
            frame.qy = float(quat.get('y', 0.0))
            frame.qz = float(quat.get('z', 0.0))
            frame.qw = float(quat.get('w', 1.0))
        
        frame.hand = 'right'
        frame.tracked = bool(right.get('tracked', True))
    
    return frame


# ==========================================
# 핵심 파이프라인 클래스
# ==========================================

class TeleopPipeline:
    """
    VR 데이터를 로봇 제어 명령으로 변환하는 파이프라인
    
    처리 순서:
    1. 워치독 체크 (연결 상태 확인)
    2. VR 기준점 설정 (초기 위치 보정)
    3. 상대 이동량 계산
    4. 좌표계 변환 및 스케일링
    5. EMA 필터 (손떨림 제거)
    6. 안전 제한 (급격한 움직임 방지)
    """
    
    def __init__(self, config: TeleopConfig):
        self.cfg = config
        
        # EMA 필터 상태
        self.last_target_pos: Optional[Tuple[float, float, float]] = None
        
        # 워치독 및 타이밍
        self.last_frame_time: float = 0.0
        
        # VR 기준점 (처음 받은 위치를 0,0,0으로 설정)
        self.vr_origin: Optional[Tuple[float, float, float]] = None
        self.origin_set: bool = False
        
        print(f"✅ TeleopPipeline 초기화 완료")
        print(f"   홈 포지션: {config.robot_home_pos}")
        print(f"   스케일: {config.pos_scale}, EMA: {config.ema_alpha}")
    
    def reset_origin(self):
        """VR 기준점 재설정 (재보정 시 사용)"""
        self.origin_set = False
        self.vr_origin = None
        self.last_target_pos = None
        print("🔄 VR 기준점 초기화됨")
    
    def process(self, frame: TeleopFrame) -> Dict[str, Any]:
        """
        TeleopFrame을 처리하여 로봇 제어 명령 생성
        
        Returns:
            dict: 사용자 코드에서 기대하는 형식의 결과
        """
        current_time = time.time()
        
        # ----------------------------------------
        # 1. 워치독 체크 (연결 끊김 감지)
        # ----------------------------------------
        if self.last_frame_time > 0:
            dt = current_time - self.last_frame_time
            if dt > self.cfg.watchdog_timeout_s:
                print(f"⚠️  워치독 발동: {dt:.2f}초 데이터 없음 - 홈으로 복귀")
                self.reset_origin()
                return self._create_home_result(frame)
        
        self.last_frame_time = current_time
        
        # ----------------------------------------
        # 2. VR 기준점 설정 (첫 프레임 또는 재연결 시)
        # ----------------------------------------
        if not self.origin_set:
            self.vr_origin = (frame.x, frame.y, frame.z)
            self.last_target_pos = self.cfg.robot_home_pos
            self.origin_set = True
            print(f"📍 VR 기준점 설정: ({frame.x:.3f}, {frame.y:.3f}, {frame.z:.3f})")
        
        # ----------------------------------------
        # 3. 상대 이동량 계산 (기준점 대비 델타)
        # ----------------------------------------
        delta_x = frame.x - self.vr_origin[0]
        delta_y = frame.y - self.vr_origin[1]
        delta_z = frame.z - self.vr_origin[2]
        
        # ----------------------------------------
        # 4. 좌표계 변환 및 스케일링
        # ----------------------------------------
        # 축 부호 변환 (VR → 로봇 좌표계)
        scaled_dx = delta_x * self.cfg.pos_scale * self.cfg.axis_sign[0]
        scaled_dy = delta_y * self.cfg.pos_scale * self.cfg.axis_sign[1]
        scaled_dz = delta_z * self.cfg.pos_scale * self.cfg.axis_sign[2]
        
        # 홈 포지션 기준 절대 좌표
        target_x = self.cfg.robot_home_pos[0] + scaled_dx
        target_y = self.cfg.robot_home_pos[1] + scaled_dy
        target_z = self.cfg.robot_home_pos[2] + scaled_dz
        
        # ----------------------------------------
        # 5. EMA 필터 (손떨림 제거)
        # ----------------------------------------
        if self.last_target_pos is not None:
            alpha = self.cfg.ema_alpha
            target_x = alpha * target_x + (1 - alpha) * self.last_target_pos[0]
            target_y = alpha * target_y + (1 - alpha) * self.last_target_pos[1]
            target_z = alpha * target_z + (1 - alpha) * self.last_target_pos[2]
        
        # ----------------------------------------
        # 6. 안전 제한 (급격한 움직임 방지)
        # ----------------------------------------
        if self.last_target_pos is not None:
            # 이전 프레임 대비 이동량 제한
            dx = target_x - self.last_target_pos[0]
            dy = target_y - self.last_target_pos[1]
            dz = target_z - self.last_target_pos[2]
            
            # 클리핑
            dx = max(-self.cfg.max_dx, min(self.cfg.max_dx, dx))
            dy = max(-self.cfg.max_dy, min(self.cfg.max_dy, dy))
            dz = max(-self.cfg.max_dz, min(self.cfg.max_dz, dz))
            
            target_x = self.last_target_pos[0] + dx
            target_y = self.last_target_pos[1] + dy
            target_z = self.last_target_pos[2] + dz
        
        # 현재 타겟 저장 (다음 프레임 기준)
        self.last_target_pos = (target_x, target_y, target_z)
        
        # ----------------------------------------
        # 7. 결과 반환 (사용자 코드 호환 형식)
        # ----------------------------------------
        return {
            "target_pose": {
                "x": round(target_x, 4),
                "y": round(target_y, 4),
                "z": round(target_z, 4),
                "qx": round(frame.qx, 4),
                "qy": round(frame.qy, 4),
                "qz": round(frame.qz, 4),
                "qw": round(frame.qw, 4),
            },
            "raw_vr_pose": {
                "x": round(frame.x, 4),
                "y": round(frame.y, 4),
                "z": round(frame.z, 4),
                "qx": round(frame.qx, 4),
                "qy": round(frame.qy, 4),
                "qz": round(frame.qz, 4),
                "qw": round(frame.qw, 4),
            },
            "status": "ok",
            "delta": {
                "dx": round(scaled_dx, 4),
                "dy": round(scaled_dy, 4),
                "dz": round(scaled_dz, 4),
            }
        }
    
    def _create_home_result(self, frame: TeleopFrame) -> Dict[str, Any]:
        """홈 포지션 결과 생성 (워치독 발동 시)"""
        return {
            "target_pose": {
                "x": self.cfg.robot_home_pos[0],
                "y": self.cfg.robot_home_pos[1],
                "z": self.cfg.robot_home_pos[2],
                "qx": 0.0, "qy": 0.0, "qz": 0.0, "qw": 1.0,
            },
            "raw_vr_pose": {
                "x": frame.x, "y": frame.y, "z": frame.z,
                "qx": frame.qx, "qy": frame.qy, "qz": frame.qz, "qw": frame.qw,
            },
            "status": "watchdog_home",
            "delta": {"dx": 0.0, "dy": 0.0, "dz": 0.0}
        }


# ==========================================
# 테스트 코드
# ==========================================
if __name__ == '__main__':
    print("=== teleop_pipeline 테스트 ===")
    
    # 사용자 설정과 동일한 구성
    cfg = TeleopConfig(
        pos_scale=0.5,
        robot_home_pos=(0.35, -0.20, 0.45),
        max_dx=0.10, max_dy=0.10, max_dz=0.10,
        ema_alpha=0.15,
        watchdog_timeout_s=0.3,
        axis_sign=(1.0, -1.0, -1.0),
    )
    
    pipeline = TeleopPipeline(cfg)
    
    # 테스트 JSON (사용자 코드에서 사용하는 flat 형식)
    test_json = {
        "hand": "right",
        "x": 0.1, "y": 0.5, "z": -0.3,
        "qx": 0.0, "qy": 0.0, "qz": 0.0, "qw": 1.0,
        "timestamp": time.time()
    }
    
    print("\n--- 테스트 실행 ---")
    frame = parse_frame_from_json(test_json)
    result = pipeline.process(frame)
    
    import json
    print(json.dumps(result, indent=2, ensure_ascii=False))
    print("\n✅ teleop_pipeline 테스트 완료!")
