#!/usr/bin/env python3
from http.server import HTTPServer, SimpleHTTPRequestHandler
import ssl
import json
import time

from teleop_pipeline import (
    TeleopConfig,
    TeleopPipeline,
    parse_frame_from_json,
)

HOST = "0.0.0.0"
PORT = 8443

# 기존 teleop pipeline 설정 (변경 없음)
cfg = TeleopConfig(
    pos_scale=0.5,
    robot_home_pos=(0.35, -0.20, 0.45),
    max_dx=0.10, max_dy=0.10, max_dz=0.10,
    ema_alpha=0.15,
    watchdog_timeout_s=0.3,
    axis_sign=(1.0, -1.0, -1.0),
)

pipeline = TeleopPipeline(cfg)


class Handler(SimpleHTTPRequestHandler):

    def do_GET(self):
        if self.path in ('/', '/index.html'):
            html = self._get_webxr_html()
            encoded = html.encode('utf-8')
            self.send_response(200)
            self.send_header('Content-Type', 'text/html; charset=utf-8')
            self.send_header('Content-Length', str(len(encoded)))
            self.end_headers()
            self.wfile.write(encoded)
        elif self.path == '/status':
            status = {"server": "running", "pipeline": "ready", "time": time.time()}
            response = json.dumps(status, indent=2).encode('utf-8')
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.send_header('Content-Length', str(len(response)))
            self.end_headers()
            self.wfile.write(response)
        else:
            self.send_response(404)
            self.end_headers()

    def do_POST(self):
        if self.path == "/pose":
            length = int(self.headers.get("Content-Length", "0"))
            raw = self.rfile.read(length)
            try:
                data = json.loads(raw.decode("utf-8"))
                frame = parse_frame_from_json(data)
                result = pipeline.process(frame)

                print("\n=== TELEOP RESULT ===")
                print(json.dumps(result, indent=2, ensure_ascii=False))

                response = json.dumps(result).encode("utf-8")
                self.send_response(200)
                self.send_header("Content-Type", "application/json")
                self.send_header("Content-Length", str(len(response)))
                self.send_header("Access-Control-Allow-Origin", "*")
                self.end_headers()
                self.wfile.write(response)

            except Exception as e:
                print("POST parse/process error:", e)
                err = json.dumps({"ok": False, "error": str(e)}).encode("utf-8")
                self.send_response(500)
                self.send_header("Content-Type", "application/json")
                self.send_header("Content-Length", str(len(err)))
                self.end_headers()
                self.wfile.write(err)

        elif self.path == "/reset":
            # 기준점 재설정 기능
            pipeline.reset_origin()
            response = json.dumps({"ok": True, "message": "Origin reset"}).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(response)))
            self.end_headers()
            self.wfile.write(response)

        else:
            self.send_response(404)
            self.end_headers()

    def log_message(self, format, *args):
        # POST 로그만 출력 (GET 로그 억제로 터미널 깔끔하게)
        if args and 'POST' in str(args[0]):
            super().log_message(format, *args)

// ============================================================
// 유틸리티 함수
// ============================================================
function log(msg) {
    const d = document.getElementById('log');
    d.innerHTML += '[' + new Date().toLocaleTimeString() + '] ' + msg + '<br>';
    d.scrollTop = d.scrollHeight;
    console.log(msg);
}

function setStatus(msg, color) {
    const s = document.getElementById('status');
    s.textContent = msg;
    s.style.background = color || 'rgba(255,255,255,0.1)';
}

// ============================================================
// 1단계: 환경 검사
// ============================================================
async function runCheck() {
    log('=== 환경 검사 시작 ===');

    const isHttps = window.location.protocol === 'https:';
    log((isHttps ? '✅' : '❌') + ' HTTPS: ' + window.location.protocol);
    if (!isHttps) { setStatus('❌ HTTPS 필수!', 'rgba(244,67,54,0.4)'); return; }

    if (!navigator.xr) {
        log('❌ navigator.xr 없음 (WebXR 미지원)');
        setStatus('❌ WebXR 미지원 브라우저', 'rgba(244,67,54,0.4)');
        return;
    }
    log('✅ navigator.xr 확인됨');

    try {
        const supported = await navigator.xr.isSessionSupported('immersive-vr');
        log((supported ? '✅' : '❌') + ' immersive-vr: ' + (supported ? '지원됨' : '미지원'));
        if (supported) {
            setStatus('✅ 모든 검사 통과 - VR 시작 가능!', 'rgba(76,175,80,0.4)');
            log('🎉 2단계 VR 시작을 눌러주세요!');
        } else {
            setStatus('❌ VR 미지원 - Guardian 설정 확인', 'rgba(244,67,54,0.4)');
        }
    } catch(e) {
        log('❌ 지원 확인 실패: ' + e.message);
    }
}

// ============================================================
// VR 세션 변수
// ============================================================
let xrSession  = null;
let xrRefSpace = null;
let gl         = null;   // ⭐️ WebGL 컨텍스트 (무한 로딩 해결 핵심)
let frameCount  = 0;
let lastSendTime = 0;
const SEND_HZ  = 30;

// ============================================================
// 2단계: VR 시작 ⭐️ 무한 로딩 해결 버전 ⭐️
// ============================================================
async function startVR() {
    log('🎯 VR 세션 시작 시도...');
    if (!navigator.xr) { log('❌ navigator.xr 없음'); return; }

    try {
        // 1) VR 세션 생성
        xrSession = await navigator.xr.requestSession('immersive-vr', {
            optionalFeatures: ['local-floor']
        });
        log('✅ VR 세션 생성 성공!');

        // ⭐️⭐️⭐️ 무한 로딩 해결의 핵심 3단계 ⭐️⭐️⭐️
        // 2) WebGL 컨텍스트 생성
        const canvas = document.createElement('canvas');
        gl = canvas.getContext('webgl', { xrCompatible: true });
        if (!gl) throw new Error('WebGL 컨텍스트 생성 실패');

        // 3) XRWebGLLayer 생성 및 세션에 연결
        await xrSession.updateRenderState({
            baseLayer: new XRWebGLLayer(xrSession, gl)
        });
        log('✅ WebGL 렌더링 레이어 연결 완료!');
        // ⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️⭐️

        // 4) 참조 공간 설정
        try {
            xrRefSpace = await xrSession.requestReferenceSpace('local-floor');
            log('✅ local-floor 참조 공간 설정');
        } catch(_) {
            xrRefSpace = await xrSession.requestReferenceSpace('local');
            log('✅ local 참조 공간 설정 (fallback)');
        }

        // 5) UI 숨기고 VR 모드 진입
        ['title','status','btn-check','btn-vr','btn-reset','log']
            .forEach(id => document.getElementById(id).style.display = 'none');
        document.body.style.background = '#111';

        setStatus('🥽 VR 모드 실행 중!', 'rgba(76,175,80,0.5)');

        // 6) 세션 종료 시 페이지 새로고침
        xrSession.addEventListener('end', () => {
            log('VR 세션 종료됨');
            location.reload();
        });

        // 7) 렌더링 루프 시작
        xrSession.requestAnimationFrame(onXRFrame);
        log('🚀 컨트롤러 추적 시작! 오른손을 움직여보세요.');

    } catch(e) {
        log('❌ VR 시작 실패: ' + e.name + ' - ' + e.message);
        const solutions = {
            'NotAllowedError' : 'Guardian 설정 또는 VR 권한 허용 필요',
            'NotSupportedError': '개발자 모드 활성화 필요',
            'SecurityError'   : 'HTTPS 주소 확인 필요',
            'InvalidStateError': '브라우저 완전 재시작 필요'
        };
        log('💡 ' + (solutions[e.name] || 'Quest 재부팅 후 재시도'));
        setStatus('❌ VR 시작 실패: ' + e.name, 'rgba(244,67,54,0.4)');
    }
}

// ============================================================
// XR 프레임 루프 (컨트롤러 데이터 추출)
// ============================================================
function onXRFrame(time, frame) {
    if (!xrSession || !gl) return;
    xrSession.requestAnimationFrame(onXRFrame);

    // ⭐️ 매 프레임마다 화면을 그려줘야 Quest가 로딩을 멈춤 ⭐️
    gl.bindFramebuffer(gl.FRAMEBUFFER,
        xrSession.renderState.baseLayer.framebuffer);
    gl.clearColor(0.05, 0.05, 0.05, 1.0);  // 어두운 회색 배경
    gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

    // 30Hz 전송 제한
    if (time - lastSendTime < 1000 / SEND_HZ) return;
    lastSendTime = time;

    // 오른손 컨트롤러 데이터 추출 및 전송
    for (const src of xrSession.inputSources) {
        if (src.handedness !== 'right' || !src.gripSpace) continue;

        const pose = frame.getPose(src.gripSpace, xrRefSpace);
        if (!pose) continue;

        const p = pose.transform.position;
        const o = pose.transform.orientation;
        sendPose('right', p.x, p.y, p.z, o.x, o.y, o.z, o.w);

        frameCount++;
        if (frameCount % 150 === 0) {
            console.log('📊 VR 데이터 전송 중:', {
                x: p.x.toFixed(3), y: p.y.toFixed(3), z: p.z.toFixed(3)
            });
        }
        break;
    }
}

// ============================================================
// 서버로 포즈 전송
// ============================================================
function sendPose(hand, x, y, z, qx, qy, qz, qw) {
    fetch('/pose', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            hand, x, y, z, qx, qy, qz, qw,
            timestamp: Date.now() / 1000
        })
    })
    .then(r => r.json())
    .then(result => {
        if (result.status === 'watchdog_home')
            console.warn('⚠️ 워치독 발동 - 홈 위치로 복귀');
    })
    .catch(e => console.debug('전송 에러:', e.message));
}

// ============================================================
// 기준점 재설정
// ============================================================
function resetOrigin() {
    log('🔄 기준점 재설정 요청...');
    fetch('/reset', { method: 'POST' })
        .then(r => r.json())
        .then(() => log('✅ 기준점 재설정 완료'))
        .catch(() => log('⚠️ 재설정 요청 실패'));
}

// ============================================================
// 초기화
// ============================================================
log('✅ 페이지 로드 완료');
log('📋 1단계 환경 검사부터 시작하세요');
setStatus('✅ 준비 완료 - 1단계 검사를 시작하세요');
</script>
</body>
</html>"""


if __name__ == '__main__':
    httpd = HTTPServer((HOST, PORT), Handler)
    httpd.socket = ssl.wrap_socket(
        httpd.socket,
        keyfile="key.pem",
        certfile="cert.pem",
        server_side=True
    )
    print(f"🚀 G1 VR Teleoperation 서버 시작")
    print(f"🌐 접속 주소: https://{HOST}:{PORT}")
    print(f"📊 상태 확인: https://{HOST}:{PORT}/status")
    print(f"🛑 종료하려면 Ctrl+C")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\n🛑 서버 종료 중...")
