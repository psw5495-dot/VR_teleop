# ROS 2 노드

class VRTeleopNode(Node):
    def __init__(self):
        super().__init__('vr_teleop_server')
        
        # Quest 3 핸드 포즈 토픽 발행
        self.right_hand_pub = self.create_publisher(
            PoseStamped, '/quest3/right_hand_pose', 10
        )
        self.left_hand_pub = self.create_publisher(
            PoseStamped, '/quest3/left_hand_pose', 10
        )
        
        self.msg_count = 0
        self.start_time = time.time()
        self.get_logger().info("✅ VR Teleoperation ROS 2 노드 시작")
        self.get_logger().info("📡 토픽: /quest3/right_hand_pose, /quest3/left_hand_pose")
        
    def publish_hand_pose(self, hand, x, y, z, qx, qy, qz, qw):
        """핸드 포즈를 ROS 2 토픽으로 발행"""
        msg = PoseStamped()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id = 'quest3_world'
        
        msg.pose.position.x = float(x)
        msg.pose.position.y = float(y)
        msg.pose.position.z = float(z)
        
        msg.pose.orientation.x = float(qx)
        msg.pose.orientation.y = float(qy)
        msg.pose.orientation.z = float(qz)
        msg.pose.orientation.w = float(qw)
        
        if hand == 'right':
            self.right_hand_pub.publish(msg)
        elif hand == 'left':
            self.left_hand_pub.publish(msg)
            
        self.msg_count += 1
        
        # 5초마다 통계 출력
        if self.msg_count % 150 == 0:  # 30Hz 기준
            elapsed = time.time() - self.start_time
            rate = self.msg_count / elapsed if elapsed > 0 else 0
            self.get_logger().info(f"📊 VR 데이터: {self.msg_count}개, 평균 {rate:.1f}Hz")

# 전역 ROS 노드
vr_node = None

def ros_spin():
    """ROS 2 노드를 별도 스레드에서 실행"""
    rclpy.init()
    global vr_node
    vr_node = VRTeleopNode()
    try:
        rclpy.spin(vr_node)
    except KeyboardInterrupt:
        pass
    finally:
        vr_node.destroy_node()
        rclpy.shutdown()

# ==========================================
# WebSocket 핸들러
# ==========================================
async def websocket_handler(request):
    """Quest 3에서 오는 WebSocket 연결 처리"""
    ws = web.WebSocketResponse()
    await ws.prepare(request)
    
    client_ip = request.remote
    print(f"📱 Quest 3 WebSocket 연결됨: {client_ip}")
    
    try:
        async for msg in ws:
            if msg.type == web.WSMsgType.TEXT:
                try:
                    data = json.loads(msg.data)
                    
                    if vr_node and 'hand' in data:
                        vr_node.publish_hand_pose(
                            data['hand'],  # 'right' 또는 'left'
                            data['x'], data['y'], data['z'],
                            data['qx'], data['qy'], data['qz'], data['qw']
                        )
                        
                except json.JSONDecodeError as e:
                    print(f"⚠️  JSON 파싱 에러: {e}")
                except KeyError as e:
                    print(f"⚠️  필수 키 누락: {e}")
                except Exception as e:
                    print(f"❌ 데이터 처리 에러: {e}")
                    
            elif msg.type == web.WSMsgType.ERROR:
                print(f"❌ WebSocket 에러: {ws.exception()}")
                break
                
    except Exception as e:
        print(f"❌ WebSocket 연결 에러: {e}")
    finally:
        print(f"📱 Quest 3 연결 해제: {client_ip}")
        
    return ws

# ==========================================
# 상태 확인 API
# ==========================================
async def status_handler(request):
    """시스템 상태 확인 API"""
    status = {
        'server': 'running',
        'ros_node': vr_node is not None,
        'message_count': vr_node.msg_count if vr_node else 0,
        'uptime': time.time() - (vr_node.start_time if vr_node else time.time())
    }
    return web.json_response(status)

var ws = null;

function connectWebSocket() {
    var protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    var wsUrl = protocol + '//' + window.location.host + '/ws';
    
    log('WebSocket 연결 시도: ' + wsUrl, 'info');
    
    try {
        ws = new WebSocket(wsUrl);
        
        ws.onopen = function() {
            log('✅ WebSocket 연결 성공!', 'ok');
            statusDiv.textContent = '✅ 서버 연결 완료 - 1단계 검사를 시작하세요';

        };
        
    } catch (error) {
        log('❌ WebSocket 생성 실패: ' + error.message, 'err');

}

# 메인 실행
# ==========================================
def main():
    parser = argparse.ArgumentParser(description='Quest 3 WebXR → ROS 2 텔레오퍼레이션 서버')
    parser.add_argument('--port', type=int, default=8443, help='HTTPS 포트 (기본: 8443)')
    parser.add_argument('--ssl-cert', required=True, help='SSL 인증서 파일 경로')
    parser.add_argument('--ssl-key', required=True, help='SSL 키 파일 경로')
    args = parser.parse_args()

    print("🚀 G1 VR Teleoperation 서버 시작 중...")
    
    # ROS 2 노드를 백그라운드 스레드에서 실행
    ros_thread = threading.Thread(target=ros_spin, daemon=True)
    ros_thread.start()
    time.sleep(2)

    # aiohttp 웹 서버 설정
    app = web.Application()
    app.router.add_get('/', index_handler)
    app.router.add_get('/ws', websocket_handler)
    app.router.add_get('/status', status_handler)

    # SSL 컨텍스트 설정
    ssl_context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ssl_context.load_cert_chain(args.ssl_cert, args.ssl_key)

    print(f"🌐 WebXR 서버 준비 완료!")
    print(f"📱 Quest 3 접속 주소: https://0.0.0.0:{args.port}")
    print(f"📊 상태 확인: https://0.0.0.0:{args.port}/status")
    print(f"🛑 종료하려면 Ctrl+C")
    
    try:
        web.run_app(app, port=args.port, ssl_context=ssl_context, print=None)
    except KeyboardInterrupt:
        print("\n🛑 서버 종료 중...")

if __name__ == '__main__':
    main()
