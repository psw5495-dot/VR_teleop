def calculate_crc_manual(data_bytes):
    """수동 CRC 계산 (SDK CRC 모듈 없을 때)"""
    crc = 0xFFFFFFFF
    for byte in data_bytes:
        crc ^= byte
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xEDB88320
            else:
                crc >>= 1
    return crc ^ 0xFFFFFFFF


def set_low_cmd_crc(low_cmd):
    """LowCmd 메시지에 CRC 설정"""
    global CRC_AVAILABLE
    
    if CRC_AVAILABLE:
        try:
            crc_calc = CRC()
            low_cmd.crc = crc_calc.Crc(low_cmd)
            return True
        except Exception as e:
            print(f"CRC 계산 실패: {e}")
    
    # 수동 CRC 계산 (근사치)
    try:
        # LowCmd 구조체를 바이트로 변환 (근사)
        data = bytearray()
        data.extend(low_cmd.mode_pr.to_bytes(1, 'little'))
        data.extend(low_cmd.mode_machine.to_bytes(1, 'little'))
        
        for motor_cmd in low_cmd.motor_cmd:
            data.extend(int(motor_cmd.mode).to_bytes(1, 'little'))
            data.extend(int(motor_cmd.q * 1000).to_bytes(4, 'little', signed=True))
            data.extend(int(motor_cmd.kp * 100).to_bytes(4, 'little', signed=True))
        
        low_cmd.crc = calculate_crc_manual(data) & 0xFFFFFFFF
        return True
    except Exception:
        low_cmd.crc = 0x12345678  # 기본 더미 CRC
        return False
