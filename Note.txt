# --- ROS 2 노드 정의 ---
class VRTeleopNode(Node):
    def __init__(self):
        super().__init__('vr_teleop_server')
        
        # Quest 3 오른손 포즈 토픽 발행
        self.right_hand_pub = self.create_publisher(
            PoseStamped, '/quest3/right_hand_pose', 10
        )
        
        # 왼손도 필요하면 추가
        self.left_hand_pub = self.create_publisher(
            PoseStamped, '/quest3/left_hand_pose', 10
        )
        
        self.get_logger().info("✅ VR Teleoperation ROS 2 노드 시작")
        self.get_logger().info("📡 토픽: /quest3/right_hand_pose, /quest3/left_hand_pose")
        
        # 통계
        self.msg_count = 0
        self.start_time = time.time()
        
    def publish_hand_pose(self, hand, x, y, z, qx, qy, qz, qw):
        """핸드 포즈를 ROS 2 토픽으로 발행"""
        msg = PoseStamped()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id = 'quest3_world'
        
        msg.pose.position.x = float(x)
        msg.pose.position.y = float(y)
        msg.pose.position.z = float(z)
        
        msg.pose.orientation.x = float(qx)
        msg.pose.orientation.y = float(qy)
        msg.pose.orientation.z = float(qz)
        msg.pose.orientation.w = float(qw)
        
        if hand == 'right':
            self.right_hand_pub.publish(msg)
        elif hand == 'left':
            self.left_hand_pub.publish(msg)
            
        self.msg_count += 1
        
        # 5초마다 통계 출력
        if self.msg_count % 150 == 0:  # 30Hz 기준
            elapsed = time.time() - self.start_time
            rate = self.msg_count / elapsed
            self.get_logger().info(f"📊 메시지: {self.msg_count}개, 평균 {rate:.1f}Hz")

# 전역 ROS 노드
vr_node = None

# --- ROS 2 스핀 스레드 ---
def ros_spin():
    """ROS 2 노드를 별도 스레드에서 실행"""
    rclpy.init()
    global vr_node
    vr_node = VRTeleopNode()
    try:
        rclpy.spin(vr_node)
    except KeyboardInterrupt:
        pass
    finally:
        vr_node.destroy_node()
        rclpy.shutdown()

# --- WebSocket 핸들러 (VR 데이터 수신) ---
async def websocket_handler(request):
    """Quest 3에서 오는 WebSocket 연결 처리"""
    ws = web.WebSocketResponse()
    await ws.prepare(request)
    
    client_ip = request.remote
    print(f"📱 Quest 3 연결됨: {client_ip}")
    
    try:
        async for msg in ws:
            if msg.type == web.WSMsgType.TEXT:
                try:
                    data = json.loads(msg.data)
                    
                    if vr_node and 'hand' in data:
                        vr_node.publish_hand_pose(
                            data['hand'],  # 'right' 또는 'left'
                            data['x'], data['y'], data['z'],
                            data['qx'], data['qy'], data['qz'], data['qw']
                        )
                        
                except json.JSONDecodeError as e:
                    print(f"⚠️  JSON 파싱 에러: {e}")
                except KeyError as e:
                    print(f"⚠️  필수 키 누락: {e}")
                except Exception as e:
                    print(f"❌ 데이터 처리 에러: {e}")
                    
            elif msg.type == web.WSMsgType.ERROR:
                print(f"❌ WebSocket 에러: {ws.exception()}")
                break
                
    except Exception as e:
        print(f"❌ WebSocket 연결 에러: {e}")
    finally:
        print(f"📱 Quest 3 연결 해제: {client_ip}")
        
    return ws

async function startXRSession(mode) {
    log(`🎯 XR 세션 시작 시도: ${mode}`, 'info');
    
    if (!navigator.xr) {
        log('❌ navigator.xr 없음! 먼저 1단계 검사를 실행하세요', 'err');
        return;
    }
    
    try {
        // 세션 옵션 (Quest Guardian 문제 방지)
        const sessionInit = {
            optionalFeatures: ['local-floor']  // bounded-floor 제거로 Guardian 문제 방지
        };
        
        log('세션 요청 전송 중...', 'info');
        const session = await navigator.xr.requestSession(mode, sessionInit);
        log('✅ XR 세션 생성 성공!', 'ok');
        
        // 렌더러에 세션 연결
        await renderer.xr.setSession(session);
        log('✅ 렌더러 연결 완료!', 'ok');
        
        // UI 전환
        document.querySelector('h1').style.display = 'none';
        document.getElementById('log-container').style.display = 'none';
        statusDiv.style.display = 'none';
        document.querySelectorAll('.btn').forEach(btn => btn.style.display = 'none');
        renderer.domElement.style.display = 'block';
        
        log('🥽 VR 모드 진입 완료! 오른손 컨트롤러를 움직여보세요', 'ok');
        
        // 세션 종료 이벤트
        session.addEventListener('end', () => {
            log('VR 세션 종료됨', 'warn');
            // UI 복원
            document.querySelector('h1').style.display = 'block';
            document.getElementById('log-container').style.display = 'block';
            statusDiv.style.display = 'block';
            document.querySelectorAll('.btn').forEach(btn => btn.style.display = 'block');
            renderer.domElement.style.display = 'none';
            
            statusDiv.textContent = 'VR 세션 종료됨 - 다시 시작하려면 버튼 클릭';
            statusDiv.style.background = 'rgba(255, 152, 0, 0.3)';
        });
        
    } catch (error) {
        log(`❌ XR 세션 실패: ${error.name}: ${error.message}`, 'err');
        
        // 구체적인 에러별 해결책
        switch (error.name) {
            case 'NotSupportedError':
                log('→ immersive-vr 미지원. 3단계 inline 모드를 시도하세요', 'warn');
                break;
            case 'SecurityError':
                log('→ 보안 오류. HTTPS 주소 재확인 필요', 'warn');
                break;
            case 'NotAllowedError':
                log('→ 사용자 권한 거부. Quest에서 VR 권한을 허용하세요', 'warn');
                break;
            case 'InvalidStateError':
                log('→ 이미 XR 세션이 실행 중입니다', 'warn');
                break;
            default:
                log('→ Quest 가디언(안전 경계) 설정을 확인하세요', 'warn');
                log('→ Quest를 재부팅 후 다시 시도해보세요', 'warn');

# --- 상태 확인 API ---
async def status_handler(request):
    """시스템 상태 확인 API"""
    status = {
        'server': 'running',
        'ros_node': vr_node is not None,
        'message_count': vr_node.msg_count if vr_node else 0,
        'uptime': time.time() - (vr_node.start_time if vr_node else time.time())
    }
    return web.json_response(status)

    args = parser.parse_args()

    print("🚀 G1 VR Teleoperation 서버 시작 중...")
    
    # ROS 2 노드를 백그라운드 스레드에서 실행
    ros_thread = threading.Thread(target=ros_spin, daemon=True)
    ros_thread.start()
    
    # 잠깐 대기 (ROS 노드 초기화)
    time.sleep(2)

    # aiohttp 웹 서버 설정
    app = web.Application()
    app.router.add_get('/', index_handler)
    app.router.add_get('/ws', websocket_handler)
    app.router.add_get('/status', status_handler)

    # SSL 컨텍스트 설정 (WebXR은 HTTPS 필수)
    ssl_context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ssl_context.load_cert_chain(args.ssl_cert, args.ssl_key)

    print(f"🌐 WebXR 서버 준비 완료!")
    print(f"📱 Quest 3 접속 주소: https://0.0.0.0:{args.port}")
    print(f"📊 상태 확인: https://0.0.0.0:{args.port}/status")
    print(f"🛑 종료하려면 Ctrl+C")
    
    try:
        web.run_app(app, port=args.port, ssl_context=ssl_context, print=None)
    except KeyboardInterrupt:
        print("\n🛑 서버 종료 중...")

if __name__ == '__main__':
    main()
