def joint_callback(self, msg):
    """ROS2 JointState → G1 DDS 명령 변환 (CRC 포함)"""
    if self.cmd_pub is None or self.low_cmd is None:
        self._log_simulation(msg)
        return
    
    try:
        matched_joints = 0
        
        # 관절 매핑 (기존 코드 동일)
        for joint_name, motor_idx in G1_RIGHT_ARM_MOTOR_IDX.items():
            if joint_name not in msg.name:
                continue
            
            j_idx = msg.name.index(joint_name)
            if j_idx >= len(msg.position):
                continue
            
            target_q = float(msg.position[j_idx])
            
            if abs(target_q) > np.pi:
                self.get_logger().warn(f'⚠️ {joint_name}: {target_q:.3f}rad 범위 초과')
                continue
            
            # 모터 명령 업데이트
            self.low_cmd.motor_cmd[motor_idx] = create_motor_cmd(
                mode=1, q=target_q, dq=0.0, tau=0.0,
                kp=self.KP, kd=self.KD, reserve=0
            )
            matched_joints += 1
        
        if matched_joints == 0:
            return
        
        # ⭐️ CRC 계산 (핵심!)
        crc_success = set_low_cmd_crc(self.low_cmd)
        
        # G1으로 DDS 전송
        self.cmd_pub.Write(self.low_cmd)
        self.cmd_count += 1
        
        # 주기적 로그 (CRC 상태 포함)
        now = time.time()
        if now - self.last_log > 5.0:
            crc_status = "✅" if crc_success else "⚠️"
            self.get_logger().info(
                f'{crc_status} DDS 전송 {self.cmd_count}회 | '
                f'매칭: {matched_joints}개 | CRC: 0x{self.low_cmd.crc:08X}'
            )
            self.last_log = now
            
    except Exception as e:
        self.get_logger().error(f'❌ joint_callback 에러: {e}')
