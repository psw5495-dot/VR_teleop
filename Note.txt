# 1. Ubuntu PC의 네트워크 정보 확인
ip addr show
ip route | grep default

# 유선 인터페이스(eth0, enp3s0 등)의 IP 주소 확인
# 예시: 192.168.1.100 ← 이 IP를 Quest3에서 사용

# 2. 작업 디렉터리 생성
mkdir -p ~/g1_teleop && cd ~/g1_teleop

# 3. Quest3 WebXR용 SSL 인증서 생성 (필수)
openssl req -x509 -newkey rsa:4096 \
    -keyout key.pem -out cert.pem \
    -days 365 -nodes \
    -subj "/CN=g1-teleop"

# 4. 방화벽 포트 개방
sudo ufw allow 8443/tcp    # WebXR HTTPS
sudo ufw allow 7400:7500/udp  # DDS 통신 (G1용)

# 5. 필요 패키지 설치
pip install unitree_sdk2py pin numpy
