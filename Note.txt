#!/usr/bin/env python3
"""
G1 로봇 팔 역기구학(IK) 제어 노드
Unitree 공식 URDF 기반 완전 구현
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from sensor_msgs.msg import JointState
import numpy as np
import pinocchio as pin
import os
import time
import threading
import xml.etree.ElementTree as ET

class G1IKNode(Node):
    def __init__(self):
        super().__init__('g1_ik_node')
        
        # URDF 로드 및 관절 분석
        self.setup_pinocchio_model()
        
        # ROS 2 토픽 설정
        self.target_sub = self.create_subscription(
            PoseStamped, '/g1/target_pose',
            self.target_callback, 10
        )
        
        self.joint_pub = self.create_publisher(
            JointState, '/g1/arm/joint_commands', 10
        )
        
        # G1 현재 상태 구독 (실제 로봇 연결 시)
        self.joint_state_sub = self.create_subscription(
            JointState, '/joint_states',
            self.joint_state_callback, 10
        )
        
        # 상태 변수
        self.current_q = pin.neutral(self.model)
        self.emergency_stop = False
        self.robot_connected = False
        self.last_log_time = time.time()
        
        # 안전 설정
        self.workspace_limits = {
            'x_min': 0.05, 'x_max': 1.0,   # 0.2→0.05 (더 가까이 허용)
            'y_min': -0.8, 'y_max': 0.8,   # ±0.5→±0.8 (더 넓게)
            'z_min': 0.1, 'z_max': 1.5     # 0.3→0.1 (더 낮게 허용)
        }
                
        # 비상 정지 모니터링
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        
        self.get_logger().info('🤖 G1 IK 노드 시작!')
        self.get_logger().info('📡 /g1/target_pose 구독 중...')
        self.get_logger().info('⚠️  s + 엔터: 비상 정지 | r + 엔터: 재개')

    def setup_pinocchio_model(self):
        """Unitree 공식 URDF 로드 및 관절 분석"""
        base_dir = os.path.dirname(os.path.abspath(__file__))
        
        # 공식 URDF 경로 (우선순위: 손 포함 → 기본)
        urdf_candidates = [
            os.path.join(base_dir, 'models', 'g1', 'g1_body29_hand14.urdf'),
            os.path.join(base_dir, 'models', 'g1', 'g1_body23.urdf'),
        ]
        
        urdf_path = None
        for path in urdf_candidates:
            if os.path.exists(path):
                urdf_path = path
                break
        
        if not urdf_path:
            self.get_logger().error('❌ G1 URDF 파일을 찾을 수 없습니다!')
            self.get_logger().error('다음 명령어로 G1 로봇에서 복사하세요:')
            self.get_logger().error('scp -r unitree@192.168.123.26:'
                                  '/home/unitree/xr_teleoperate/assets/g1 '
                                  '~/TeleVision/models/')
            exit(1)
        
        # Pinocchio 모델 빌드
        try:
            self.model = pin.buildModelFromUrdf(urdf_path)
            self.data = self.model.createData()
            self.get_logger().info(f'✅ G1 공식 URDF 로드 성공: {os.path.basename(urdf_path)}')
            self.get_logger().info(f'   총 관절 수: {self.model.nq}')
            self.get_logger().info(f'   총 프레임 수: {self.model.nframes}')
        except Exception as e:
            self.get_logger().error(f'❌ URDF 파싱 실패: {e}')
            exit(1)
        
        # 오른팔 관절 자동 파싱
        self.right_arm_joints = self._parse_right_arm_joints(urdf_path)
        if self.right_arm_joints:
            self.get_logger().info(f'✅ 오른팔 관절 ({len(self.right_arm_joints)}개):')
            for i, name in enumerate(self.right_arm_joints):
                self.get_logger().info(f'   [{i}] {name}')
        else:
            self.get_logger().error('❌ 오른팔 관절을 찾을 수 없습니다!')
        
        # End-effector 프레임 설정
        self.ee_frame_id = self._find_ee_frame()

        def _parse_right_arm_joints(self, urdf_path):
            """오른팔 관절만 정확히 추출 (왼팔 완전 제외)"""
            try:
                tree = ET.parse(urdf_path)
                root = tree.getroot()
                
                # ⭐️ G1 오른팔 표준 관절 이름 (순서 중요)
                target_joints = [
                    'right_shoulder_pitch_joint',
                    'right_shoulder_roll_joint', 
                    'right_shoulder_yaw_joint',
                    'right_elbow_pitch_joint',    # elbow_joint → elbow_pitch_joint
                    'right_wrist_roll_joint',
                    'right_wrist_pitch_joint',
                    'right_wrist_yaw_joint'
                ]
                
                # URDF에서 실제 존재하는 관절만 선택
                valid_joints = []
                for target_name in target_joints:
                    for joint in root.findall('joint'):
                        joint_name = joint.get('name', '')
                        joint_type = joint.get('type', '')
                        
                        # 정확한 이름 매치 + revolute/continuous 타입만
                        if (joint_name == target_name and 
                            joint_type in ['revolute', 'continuous']):
                            valid_joints.append(joint_name)
                            break
                
                self.get_logger().info(f'✅ 필터링된 오른팔 관절 ({len(valid_joints)}개):')
                for i, name in enumerate(valid_joints):
                    self.get_logger().info(f'   [{i}] {name}')
                    
                return valid_joints
                
            except Exception as e:
                self.get_logger().error(f'URDF 관절 파싱 에러: {e}')
                return []

            def _find_ee_frame(self):
                """End-effector 프레임 자동 탐색"""
                # Unitree 공식 모델의 일반적인 end-effector 프레임명
                ee_candidates = [
                    'right_wrist_yaw_link',
                    'right_wrist_pitch_link', 
                    'right_wrist_roll_link',
                    'right_hand_base_link',
                    'right_hand_link',
                    'right_elbow_pitch_link'
                ]
                
                for name in ee_candidates:
                    if self.model.existFrame(name):
                        frame_id = self.model.getFrameId(name)
                        self.get_logger().info(f'✅ End-effector 프레임: {name} (ID: {frame_id})')
                        return frame_id
                
                # 모든 프레임 출력 (디버깅용)
                self.get_logger().warn('⚠️ 표준 end-effector 프레임을 찾지 못함')
                self.get_logger().info('사용 가능한 프레임들:')
                for i in range(min(20, self.model.nframes)):
                    frame_name = self.model.frames[i].name
                    self.get_logger().info(f'   [{i}] {frame_name}')
                
                # 마지막 프레임 사용
                frame_id = self.model.nframes - 1
                name = self.model.frames[frame_id].name
                self.get_logger().warn(f'⚠️ 마지막 프레임 사용: {name}')
                return frame_id

            def keyboard_monitor(self):
                """비상 정지 키보드 모니터링"""
                try:
                    while rclpy.ok():
                        key = input().strip().lower()
                        if key in ['s', 'stop', ' ']:
                            self.emergency_stop = True
                            self.get_logger().warn('🛑 비상 정지 활성화!')
                        elif key in ['r', 'start']:
                            self.emergency_stop = False
                            self.get_logger().info('▶️ 텔레오퍼레이션 재개')
                except:
                    pass

            def joint_state_callback(self, msg):
                """G1 현재 관절 상태 업데이트"""
                if not self.robot_connected:
                    self.robot_connected = True
                    self.get_logger().info('✅ G1 로봇 연결됨!')
                
                # 현재 관절 각도를 IK 초기값으로 업데이트
                try:
                    if len(msg.position) >= len(self.right_arm_joints):
                        # 오른팔 관절 매핑 (실제 G1 토픽 구조에 맞게 수정 필요)
                        for i, joint_name in enumerate(self.right_arm_joints):
                            if joint_name in msg.name:
                                joint_idx = msg.name.index(joint_name)
                                self.current_q[i] = msg.position[joint_idx]
                except Exception as e:
                    self.get_logger().debug(f'관절 상태 업데이트 에러: {e}')

            def check_workspace_limits(self, target_pos):
                """작업 공간 안전 제한 확인"""
                x, y, z = target_pos
                
                limits = self.workspace_limits
                if not (limits['x_min'] <= x <= limits['x_max']):
                    return False, f"x 좌표 범위 초과: {x:.3f} (허용: {limits['x_min']}-{limits['x_max']})"
                if not (limits['y_min'] <= y <= limits['y_max']):
                    return False, f"y 좌표 범위 초과: {y:.3f} (허용: {limits['y_min']}-{limits['y_max']})"
                if not (limits['z_min'] <= z <= limits['z_max']):
                    return False, f"z 좌표 범위 초과: {z:.3f} (허용: {limits['z_min']}-{limits['z_max']})"
                
                return True, "OK"

            def solve_ik(self, target_pos):
                """Pinocchio 기반 IK 계산"""
                q = self.current_q.copy()
                
                eps = 1e-3
                max_iter = 50
                dt = 0.1
                damping = 1e-4
                
                for iteration in range(max_iter):
                    # Forward Kinematics
                    pin.forwardKinematics(self.model, self.data, q)
                    pin.updateFramePlacements(self.model, self.data)
                    
                    # 현재 end-effector 위치
                    current_pos = self.data.oMf[self.ee_frame_id].translation
                    error = target_pos - current_pos
                    
                    # 수렴 확인
                    if np.linalg.norm(error) < eps:
                        return q, True
                    
                    # Jacobian 계산 (위치만 사용)
                    J_full = pin.computeFrameJacobian(
                        self.model, self.data, q, self.ee_frame_id,
                        pin.ReferenceFrame.LOCAL_WORLD_ALIGNED
                    )
                    J_pos = J_full[:3, :]  # 위치 부분만
                    
                    # Damped Least Squares
                    JJT = J_pos @ J_pos.T
                    dq = J_pos.T @ np.linalg.solve(JJT + damping * np.eye(3), error)
                    
                    # 관절 각도 업데이트
                    q = q + dt * dq
                    
                    # 관절 한계 적용
                    q = np.clip(q, self.model.lowerPositionLimit, 
                               self.model.upperPositionLimit)
                
                # 수렴 실패하더라도 근사해 반환
                return q, False

            def target_callback(self, msg):
                """VR 목표 좌표를 관절 각도로 변환"""
                if self.emergency_stop:
                    return
                
                try:
                    target_pos = np.array([
                        msg.pose.position.x,
                        msg.pose.position.y,
                        msg.pose.position.z
                    ])
                    
                    # 작업 공간 제한 확인
                    is_safe, safety_msg = self.check_workspace_limits(target_pos)
                    if not is_safe:
                        self.get_logger().warn(f'⚠️ 안전 제한: {safety_msg}')
                        return
                    
                    # IK 계산
                    q_target, converged = self.solve_ik(target_pos)
                    
                    if q_target is not None:
                        self.publish_joint_command(q_target)
                        self.current_q = q_target
                        
                        # 수렴 상태 로그
                        now = time.time()
                        if now - self.last_log_time > 2.0:  # 2초마다
                            status = "수렴됨" if converged else "근사해"
                            self.get_logger().info(f'✅ IK {status} → 관절 명령 전송')
                            self.last_log_time = now
                        
                except Exception as e:
                    self.get_logger().error(f'목표 좌표 처리 에러: {e}')

            def publish_joint_command(self, q):
                """관절 명령 발행"""
                if not self.right_arm_joints:
                    self.get_logger().error('오른팔 관절 이름이 설정되지 않았습니다!')
                    return
                
                msg = JointState()
                msg.header.stamp = self.get_clock().now().to_msg()
                msg.name = self.right_arm_joints
                
                # 오른팔 관절 수만큼 각도 추출
                n_joints = len(self.right_arm_joints)
                msg.position = q[:n_joints].tolist()
                msg.velocity = [0.0] * n_joints
                
                self.joint_pub.publish(msg)

def main():
    rclpy.init()
    node = G1IKNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        node.get_logger().info('🛑 G1 IK 노드 종료')
    finally:
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
