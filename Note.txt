#!/usr/bin/env python3
# ~/TeleVision/g1_bridge_node.py
"""
Meta Quest3 → G1 Teleoperation 브릿지 (완전 수정 버전)
핵심: 500Hz 제어 루프 + 올바른 관절 인덱스 + 증분 제어 + CRC 계산
"""

import sys
import time
import numpy as np
import threading
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState

# ── Unitree SDK2 임포트 ────────────────────────────────────────
from unitree_sdk2py.core.channel import (
    ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
)
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_, MotorCmd_
from unitree_sdk2py.utils.thread import RecurrentThread
from unitree_sdk2py.comm.motion_switcher.motion_switcher_client import MotionSwitcherClient

# ⭐️ CRC 모듈 로드 (로봇 제어에 필수)
CRC_AVAILABLE = False
try:
    from unitree_sdk2py.utils.crc import CRC
    CRC_AVAILABLE = True
    print("✅ CRC 모듈 로드 성공")
except ImportError:
    try:
        from unitree_sdk2py.core.crc import CRC
        CRC_AVAILABLE = True
        print("✅ CRC 모듈 로드 성공 (core 경로)")
    except ImportError:
        print("⚠️ CRC 모듈 없음 - 수동 계산 사용")

# ── G1 로봇 사양 ──────────────────────────────────────────────
G1_NUM_MOTOR = 29

# 공식 예제의 Kp/Kd 값
Kp = [
    60, 60, 60, 100, 40, 40,      # left leg
    60, 60, 60, 100, 40, 40,      # right leg  
    60, 40, 40,                   # waist
    40, 40, 40, 40, 40, 40, 40,   # left arm (15~21)
    40, 40, 40, 40, 40, 40, 40    # right arm (22~28)
]

Kd = [
    1, 1, 1, 2, 1, 1,     # left leg
    1, 1, 1, 2, 1, 1,     # right leg
    1, 1, 1,              # waist
    1, 1, 1, 1, 1, 1, 1,  # left arm
    1, 1, 1, 1, 1, 1, 1   # right arm
]

class Mode:
    PR = 0
    AB = 1

class G1JointIndex:
    LeftShoulderPitch = 15
    LeftShoulderRoll = 16
    LeftShoulderYaw = 17
    LeftElbow = 18
    LeftWristRoll = 19
    LeftWristPitch = 20
    LeftWristYaw = 21
    
    RightShoulderPitch = 22
    RightShoulderRoll = 23
    RightShoulderYaw = 24
    RightElbow = 25
    RightWristRoll = 26
    RightWristPitch = 27
    RightWristYaw = 28

ROS_TO_G1_INDEX = {
    'right_shoulder_pitch_joint': G1JointIndex.RightShoulderPitch,
    'right_shoulder_roll_joint' : G1JointIndex.RightShoulderRoll,
    'right_shoulder_yaw_joint'  : G1JointIndex.RightShoulderYaw,
    'right_elbow_pitch_joint'   : G1JointIndex.RightElbow,
    'right_wrist_roll_joint'    : G1JointIndex.RightWristRoll,
    'right_wrist_pitch_joint'   : G1JointIndex.RightWristPitch,
    'right_wrist_yaw_joint'     : G1JointIndex.RightWristYaw,
}

LEFT_ARM_HOME = {
    G1JointIndex.LeftShoulderPitch: 0.0,
    G1JointIndex.LeftShoulderRoll:  0.3,
    G1JointIndex.LeftShoulderYaw:   0.0,
    G1JointIndex.LeftElbow:         0.5,
    G1JointIndex.LeftWristRoll:     0.0,
    G1JointIndex.LeftWristPitch:    0.0,
    G1JointIndex.LeftWristYaw:      0.0,
}

# ── 헬퍼 함수 ──────────────────────────────────────────────────
def create_motor_cmd(mode=0, q=0.0, dq=0.0, tau=0.0, kp=0.0, kd=0.0, reserve=0):
    return MotorCmd_(
        mode=int(mode), q=float(q), dq=float(dq),
        tau=float(tau), kp=float(kp), kd=float(kd), reserve=int(reserve)
    )

def create_empty_low_cmd():
    motor_cmds = [create_motor_cmd() for _ in range(35)]
    return LowCmd_(
        mode_pr=0, mode_machine=0, motor_cmd=motor_cmds,
        reserve=[0, 0, 0, 0], crc=0
    )

def calculate_crc_manual(data_bytes):
    """수동 CRC32 계산"""
    crc = 0xFFFFFFFF
    for byte in data_bytes:
        crc ^= byte
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xEDB88320
            else:
                crc >>= 1
    return crc ^ 0xFFFFFFFF

def set_low_cmd_crc(low_cmd):
    """⭐️ LowCmd에 올바른 CRC 설정 (로봇 제어 필수)"""
    if CRC_AVAILABLE:
        try:
            crc_calc = CRC()
            low_cmd.crc = crc_calc.Crc(low_cmd)
            return True
        except Exception:
            pass
    
    # 수동 CRC 계산
    try:
        data = bytearray()
        data.extend(int(low_cmd.mode_pr).to_bytes(1, 'little'))
        data.extend(int(low_cmd.mode_machine).to_bytes(1, 'little'))
        for motor_cmd in low_cmd.motor_cmd:
            data.extend(int(motor_cmd.mode).to_bytes(1, 'little'))
            data.extend(int(motor_cmd.q * 1000).to_bytes(4, 'little', signed=True))
            data.extend(int(motor_cmd.kp * 100).to_bytes(4, 'little', signed=True))
        low_cmd.crc = calculate_crc_manual(data) & 0xFFFFFFFF
        return True
    except Exception:
        low_cmd.crc = 0x12345678
        return False


# ── 메인 브릿지 노드 ───────────────────────────────────────────
class G1TeleopBridge(Node):
    def __init__(self):
        super().__init__('g1_teleop_bridge')
        
        # 제어 변수
        self.control_dt_ = 0.002
        self.init_duration_ = 3.0
        self.time_ = 0.0
        
        self.mode_pr_ = Mode.PR
        self.mode_machine_ = 0
        self.update_mode_machine_ = False
        
        # Unitree 메시지
        self.low_cmd = create_empty_low_cmd()
        self.low_state = None
        
        # 텔레오퍼레이션 상태
        self.initial_q = [0.0] * 35
        self.target_q = [0.0] * 35
        self.current_smooth_q = [0.0] * 35
        
        # ⭐️ 증분 제어 파라미터 (움찔함 방지)
        self.MAX_DELTA = 0.0005       # 매우 작은 변화량으로 시작
        self.SOFT_KP_RATIO = 0.6      # 부드러운 게인
        self.SOFT_KD_RATIO = 0.8
        
        self.initialized_smooth = False
        self.initial_posture_captured = False
        self.teleop_active = False
        self.emergency_stop = False
        
        # 통계
        self.cmd_count = 0
        self.last_log = time.time()
        
        # ROS2 구독
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', 
            self.ros_joint_callback, 10
        )
        
        self.get_logger().info("🤖 G1 Teleop Bridge 초기화 완료")
        self.get_logger().info("⚠️  로봇 주변 안전 확인 후 시작하세요!")

    def Init(self):
        """Unitree 예제와 동일한 초기화 구조"""
        self.get_logger().info("🔍 G1 모션 제어 권한 획득 중...")
        self.msc = MotionSwitcherClient()
        self.msc.SetTimeout(5.0)
        self.msc.Init()

        max_retry = 3
        for attempt in range(max_retry):
            try:
                status, result = self.msc.CheckMode()
                if result is None:
                    if attempt < max_retry - 1:
                        time.sleep(2)
                        continue
                    else:
                        self.get_logger().warn("   MotionSwitcher 건너뜀")
                        break
                
                if result.get('name', ''):
                    self.get_logger().info(f"   모션 '{result['name']}' 해제 중...")
                    self.msc.ReleaseMode()
                    time.sleep(1)
                else:
                    self.get_logger().info("   ✅ 이미 Low-level 모드")
                    break
            except Exception as e:
                self.get_logger().warn(f"   CheckMode 예외: {e}")
                break

        # DDS 채널 초기화
        self.lowcmd_publisher_ = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.lowcmd_publisher_.Init()

        self.lowstate_subscriber = ChannelSubscriber("rt/lowstate", LowState_)
        self.lowstate_subscriber.Init(self.LowStateHandler, 10)
        
        self.get_logger().info("✅ DDS 채널 초기화 완료")

    def Start(self):
        """제어 스레드 시작"""
        self.lowCmdWriteThreadPtr = RecurrentThread(
            interval=self.control_dt_,
            target=self.LowCmdWrite,
            name="g1_control"
        )
        
        self.get_logger().info("⏳ G1 로봇 상태 수신 대기 중...")
        timeout, elapsed = 10.0, 0.0
        
        while not self.update_mode_machine_:
            time.sleep(0.5)
            elapsed += 0.5
            if elapsed % 2 == 0:
                self.get_logger().info(f"   대기 중... ({elapsed:.0f}s)")
            if elapsed >= timeout:
                self.get_logger().warn("   타임아웃 - 기본값으로 진행")
                self.mode_machine_ = 0
                self.update_mode_machine_ = True
                break

        self.get_logger().info(f"✅ mode_machine: {self.mode_machine_}")
        self.get_logger().info("🚀 500Hz 제어 루프 시작!")
        
        # 키보드 모니터링 시작
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        self.lowCmdWriteThreadPtr.Start()

    # ⭐️ 들여쓰기 수정: 클래스 메서드로 올바르게 정의
    def LowStateHandler(self, msg: LowState_):
        """G1 로봇 현재 상태 수신 (500Hz)"""
        self.low_state = msg

        if not self.update_mode_machine_:
            self.mode_machine_ = msg.mode_machine
            
            # 현재 관절 각도로 안전하게 초기화
            for i in range(min(35, len(msg.motor_state))):
                actual_q = msg.motor_state[i].q
                self.initial_q[i] = actual_q
                self.target_q[i] = actual_q
                self.current_smooth_q[i] = actual_q
            
            self.initialized_smooth = True
            self.initial_posture_captured = True
            self.update_mode_machine_ = True

    def ros_joint_callback(self, msg):
        """ROS2에서 VR IK 목표 각도 수신"""
        if not self.initial_posture_captured or self.emergency_stop:
            return
            
        matched_joints = 0
        for joint_name, g1_idx in ROS_TO_G1_INDEX.items():
            if joint_name in msg.name:
                idx_in_msg = msg.name.index(joint_name)
                target_angle = float(msg.position[idx_in_msg])
                if abs(target_angle) > np.pi:
                    self.get_logger().warn(f"⚠️ {joint_name}: {target_angle:.3f}rad 범위 초과")
                    continue
                self.target_q[g1_idx] = target_angle
                matched_joints += 1
        
        # 통계 로그
        now = time.time()
        if now - self.last_log > 3.0:
            self.get_logger().info(f"📡 VR 명령 수신: {matched_joints}개 관절 업데이트")
            self.last_log = now

    def LowCmdWrite(self):
        """500Hz 메인 제어 루프"""
        if not self.initial_posture_captured or self.low_state is None:
            return

        self.time_ += self.control_dt_

        if self.emergency_stop:
            self._send_hold_position()
        elif self.time_ < self.init_duration_:
            self._stage_init_home()
        else:
            if not self.teleop_active:
                self.teleop_active = True
                self.get_logger().info("🎮 VR 텔레오퍼레이션 모드 활성화!")
                self.get_logger().info("   Quest3 컨트롤러를 천천히 움직여보세요")
            self._stage_teleop()

        # ⭐️ CRC 계산 후 전송 (필수!)
        crc_ok = set_low_cmd_crc(self.low_cmd)
        self.lowcmd_publisher_.Write(self.low_cmd)
        self.cmd_count += 1
        
        # 주기적 상태 로그
        if self.cmd_count % 2500 == 0:  # 5초마다
            crc_status = "✅" if crc_ok else "⚠️"
            self.get_logger().info(f'{crc_status} DDS 전송 {self.cmd_count}회 | CRC: 0x{self.low_cmd.crc:08X}')

    # ⭐️ 들여쓰기 수정: 클래스 메서드로 올바르게 정의
    def _stage_init_home(self):
        """Stage 1: 안전한 홈 포지션으로 이동"""
        ratio = np.clip(self.time_ / self.init_duration_, 0.0, 1.0)
        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        # 모든 관절 기본 설정
        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]
            self.low_cmd.motor_cmd[i].q = self.initial_q[i]

        # 왼팔만 홈 포지션으로 부드럽게 이동
        for motor_idx, home_q in LEFT_ARM_HOME.items():
            start_q = self.initial_q[motor_idx]
            current_q = (1.0 - ratio) * start_q + ratio * home_q
            self.low_cmd.motor_cmd[motor_idx].q = current_q
            self.current_smooth_q[motor_idx] = current_q
            self.target_q[motor_idx] = current_q

    # ⭐️ 들여쓰기 수정: 클래스 메서드로 올바르게 정의
    def _stage_teleop(self):
        """Stage 2: VR 컨트롤러로 오른팔 제어 (아주 조금씩 증분 이동)"""
        if not self.initialized_smooth:
            return
            
        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        # 오른팔 관절 인덱스 (22~28번)
        right_arm_indices = [22, 23, 24, 25, 26, 27, 28]

        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0

            if i in right_arm_indices:
                # ⭐️ 오른팔: MAX_DELTA 증분 제어 (움찔함 방지)
                target_q = self.target_q[i]
                current_smooth_q = self.current_smooth_q[i]
                diff = target_q - current_smooth_q
                
                # 매우 작은 변화량으로 제한
                if abs(diff) > self.MAX_DELTA:
                    step = np.sign(diff) * self.MAX_DELTA
                    self.current_smooth_q[i] = current_smooth_q + step
                else:
                    self.current_smooth_q[i] = target_q

                # 부드러워진 각도를 모터에 전달
                self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                
                # 게인을 낮춰서 더 부드럽게
                self.low_cmd.motor_cmd[i].kp = Kp[i] * self.SOFT_KP_RATIO
                self.low_cmd.motor_cmd[i].kd = Kd[i] * self.SOFT_KD_RATIO
            else:
                # 다리/허리/왼팔: 현재 위치 유지
                self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                self.low_cmd.motor_cmd[i].kp = Kp[i]
                self.low_cmd.motor_cmd[i].kd = Kd[i]

    # ⭐️ 들여쓰기 수정: 클래스 메서드로 올바르게 정의
    def _send_hold_position(self):
        """비상 정지: 현재 위치 유지"""
        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]

    # ⭐️ 들여쓰기 수정: 클래스 메서드로 올바르게 정의
    def keyboard_monitor(self):
        """키보드 비상 정지 및 속도 조절"""
        print("\n⌨️  키보드 제어:")
        print("   s + Enter: 비상 정지")
        print("   r + Enter: 제어 재개")
        print("   + + Enter: 속도 증가 (x1.5)")
        print("   - + Enter: 속도 감소 (÷1.5)")
        print(f"   현재 MAX_DELTA: {self.MAX_DELTA:.5f} rad/tick\n")
        
        try:
            while True:
                key = input().strip().lower()
                if key in ['s', 'stop']:
                    self.emergency_stop = True
                    self.get_logger().warn('🛑 비상 정지!')
                elif key in ['r', 'resume']:
                    self.emergency_stop = False
                    self.get_logger().info('▶️ 제어 재개')
                elif key == '+':
                    self.MAX_DELTA = min(self.MAX_DELTA * 1.5, 0.01)  # 최대 0.01 라디안
                    self.get_logger().info(f'⚡ 속도 증가: MAX_DELTA={self.MAX_DELTA:.5f}')
                elif key == '-':
                    self.MAX_DELTA = max(self.MAX_DELTA / 1.5, 0.0001)  # 최소 0.0001 라디안
                    self.get_logger().info(f'🐢 속도 감소: MAX_DELTA={self.MAX_DELTA:.5f}')
        except (EOFError, KeyboardInterrupt):
            pass


def main():
    print("=" * 70)
    print("🤖 Meta Quest3 → G1 Teleoperation 브릿지 시작")
    print("⚠️  주의: 로봇 주변에 장애물이 없는지 확인하세요!")
    print("=" * 70)
    
    if len(sys.argv) > 1:
        iface = sys.argv[1]
        print(f"🌐 지정된 네트워크 인터페이스: {iface}")
    else:
        print("❌ 네트워크 인터페이스를 인자로 전달해주세요.")
        print("   사용법: python3 g1_bridge_node.py enx80691ad9f138")
        sys.exit(1)

    # DDS 초기화
    ChannelFactoryInitialize(0, iface)
    time.sleep(0.5)

    # ROS2 노드 및 G1 제어 초기화
    rclpy.init()
    bridge_node = G1TeleopBridge()
    
    try:
        bridge_node.Init()
        bridge_node.Start()
        rclpy.spin(bridge_node)
    except KeyboardInterrupt:
        print("\n🛑 브릿지 노드 종료 중...")
    finally:
        bridge_node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
