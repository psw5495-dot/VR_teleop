#!/usr/bin/env python3
# ~/TeleVision/g1_bridge_node.py
"""
Meta Quest3 → G1 Teleoperation 순수 브릿지
핵심: 실행 시 스스로 움직이지 않고, VR 명령 시에만 부드럽게 이동
"""

import sys
import time
import numpy as np
import threading
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState

from unitree_sdk2py.core.channel import (
    ChannelPublisher, ChannelSubscriber, ChannelFactoryInitialize
)
from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_, LowState_, MotorCmd_
from unitree_sdk2py.utils.thread import RecurrentThread
from unitree_sdk2py.comm.motion_switcher.motion_switcher_client import MotionSwitcherClient

# ── CRC 모듈 로드 ──────────────────────────────────────────────
CRC_AVAILABLE = False
try:
    from unitree_sdk2py.utils.crc import CRC
    CRC_AVAILABLE = True
    print("✅ CRC 모듈 로드 성공")
except ImportError:
    try:
        from unitree_sdk2py.core.crc import CRC
        CRC_AVAILABLE = True
        print("✅ CRC 모듈 로드 성공 (core 경로)")
    except ImportError:
        print("⚠️ CRC 모듈 없음 - 수동 계산 사용")

# ── G1 로봇 사양 ──────────────────────────────────────────────
G1_NUM_MOTOR = 29

Kp = [
    60, 60, 60, 100, 40, 40,      # left leg
    60, 60, 60, 100, 40, 40,      # right leg  
    60, 40, 40,                   # waist
    40, 40, 40, 40, 40, 40, 40,   # left arm (15~21)
    40, 40, 40, 40, 40, 40, 40    # right arm (22~28)
]

Kd = [
    1, 1, 1, 2, 1, 1,     # left leg
    1, 1, 1, 2, 1, 1,     # right leg
    1, 1, 1,              # waist
    1, 1, 1, 1, 1, 1, 1,  # left arm
    1, 1, 1, 1, 1, 1, 1   # right arm
]

class Mode:
    PR = 0

# ⭐️ 오른팔만 제어 (22~28번)
RIGHT_ARM_INDICES = [22, 23, 24, 25, 26, 27, 28]

ROS_TO_G1_INDEX = {
    'right_shoulder_pitch_joint': 22,
    'right_shoulder_roll_joint' : 23,
    'right_shoulder_yaw_joint'  : 24,
    'right_elbow_pitch_joint'   : 25,
    'right_wrist_roll_joint'    : 26,
    'right_wrist_pitch_joint'   : 27,
    'right_wrist_yaw_joint'     : 28,
}

# ── 헬퍼 함수 ──────────────────────────────────────────────────
def create_motor_cmd(mode=0, q=0.0, dq=0.0, tau=0.0, kp=0.0, kd=0.0, reserve=0):
    return MotorCmd_(
        mode=int(mode), q=float(q), dq=float(dq),
        tau=float(tau), kp=float(kp), kd=float(kd), reserve=int(reserve)
    )

def create_empty_low_cmd():
    motor_cmds = [create_motor_cmd() for _ in range(35)]
    return LowCmd_(
        mode_pr=0, mode_machine=0, motor_cmd=motor_cmds,
        reserve=[0, 0, 0, 0], crc=0
    )

def calculate_crc_manual(data_bytes):
    crc = 0xFFFFFFFF
    for byte in data_bytes:
        crc ^= byte
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ 0xEDB88320
            else:
                crc >>= 1
    return crc ^ 0xFFFFFFFF

def set_low_cmd_crc(low_cmd):
    """CRC 계산 및 설정"""
    if CRC_AVAILABLE:
        try:
            crc_calc = CRC()
            low_cmd.crc = crc_calc.Crc(low_cmd)
            return True
        except Exception:
            pass
    
    try:
        data = bytearray()
        data.extend(int(low_cmd.mode_pr).to_bytes(1, 'little'))
        data.extend(int(low_cmd.mode_machine).to_bytes(1, 'little'))
        for motor_cmd in low_cmd.motor_cmd:
            data.extend(int(motor_cmd.mode).to_bytes(1, 'little'))
            data.extend(int(motor_cmd.q * 1000).to_bytes(4, 'little', signed=True))
            data.extend(int(motor_cmd.kp * 100).to_bytes(4, 'little', signed=True))
        low_cmd.crc = calculate_crc_manual(data) & 0xFFFFFFFF
        return True
    except Exception:
        low_cmd.crc = 0x12345678
        return False


class G1TeleopBridge(Node):
    def __init__(self):
        super().__init__('g1_teleop_bridge')
        
        # 제어 변수
        self.control_dt_ = 0.002  # 500Hz
        self.mode_machine_ = 0
        self.update_mode_machine_ = False
        
        # 메시지 객체
        self.low_cmd = create_empty_low_cmd()
        self.low_state = None
        
        # ⭐️ 관절 각도 추적 (핵심: 모두 현재 위치로 초기화)
        self.target_q = [0.0] * 35          # VR에서 받은 목표 각도
        self.current_smooth_q = [0.0] * 35  # 증분 제어용 부드러운 각도
        
        # 증분 제어 파라미터 (부드러운 움직임)
        self.MAX_DELTA = 0.0005              # 매우 작은 변화량
        self.SOFT_KP_RATIO = 0.6             # 부드러운 게인
        self.SOFT_KD_RATIO = 0.8
        
        # 상태 추적
        self.initial_posture_captured = False
        self.emergency_stop = False
        self.vr_active = False               # VR 데이터 수신 여부
        
        # 통계
        self.cmd_count = 0
        self.last_log = time.time()
        self.last_vr_time = 0.0
        
        # ROS2 구독
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', 
            self.ros_joint_callback, 10
        )
        
        self.get_logger().info("🤖 G1 순수 브릿지 노드 초기화 완료")
        self.get_logger().info("⭐ 특징: 실행 시 로봇이 스스로 움직이지 않음")

    def Init(self):
        """DDS 초기화"""
        self.get_logger().info("🔍 G1 모션 제어 권한 획득 중...")
        self.msc = MotionSwitcherClient()
        self.msc.SetTimeout(5.0)
        self.msc.Init()

        for attempt in range(3):
            try:
                status, result = self.msc.CheckMode()
                if result is None:
                    time.sleep(2)
                    continue
                if result.get('name', ''):
                    self.get_logger().info(f"   모션 '{result['name']}' 해제 중...")
                    self.msc.ReleaseMode()
                    time.sleep(1)
                else:
                    self.get_logger().info("   ✅ 이미 Low-level 모드")
                    break
            except Exception as e:
                self.get_logger().warn(f"   CheckMode 예외: {e}")
                break

        # DDS 채널 생성
        self.lowcmd_publisher_ = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.lowcmd_publisher_.Init()
        self.lowstate_subscriber = ChannelSubscriber("rt/lowstate", LowState_)
        self.lowstate_subscriber.Init(self.LowStateHandler, 10)
        
        self.get_logger().info("✅ DDS 채널 초기화 완료")

    def Start(self):
        """제어 루프 시작"""
        self.lowCmdWriteThreadPtr = RecurrentThread(
            interval=self.control_dt_, target=self.LowCmdWrite, name="g1_control"
        )
        
        # G1 상태 수신 대기
        self.get_logger().info("⏳ G1 로봇 상태 수신 대기 중...")
        timeout, elapsed = 10.0, 0.0
        while not self.update_mode_machine_:
            time.sleep(0.5)
            elapsed += 0.5
            if elapsed % 2 == 0:
                self.get_logger().info(f"   대기 중... ({elapsed:.0f}s)")
            if elapsed >= timeout:
                self.get_logger().warn("   타임아웃 - 기본값으로 진행")
                self.mode_machine_ = 0
                self.update_mode_machine_ = True
                break

        self.get_logger().info(f"✅ mode_machine: {self.mode_machine_}")
        self.get_logger().info("🚀 500Hz 제어 루프 시작!")
        self.get_logger().info("📋 현재 상태: 로봇 완전 정지 (VR 명령 대기 중)")
        
        # 키보드 모니터링 시작
        threading.Thread(target=self.keyboard_monitor, daemon=True).start()
        self.lowCmdWriteThreadPtr.Start()

    def LowStateHandler(self, msg: LowState_):
        """⭐️ G1 현재 상태 수신 - 최초 수신 시 현재 각도로 완전 초기화"""
        self.low_state = msg

        if not self.update_mode_machine_:
            self.mode_machine_ = msg.mode_machine
            
            # ⭐️ 핵심: 로봇의 현재 실제 각도를 목표 각도로 설정
            # 이렇게 하면 target_q == current_smooth_q 이므로 로봇이 전혀 움직이지 않음
            for i in range(min(35, len(msg.motor_state))):
                actual_q = msg.motor_state[i].q
                self.target_q[i] = actual_q
                self.current_smooth_q[i] = actual_q
            
            self.initial_posture_captured = True
            self.update_mode_machine_ = True
            self.get_logger().info("✅ 현재 자세로 초기화 완료 (움직임 없음)")

    def ros_joint_callback(self, msg):
        """⭐️ VR에서 새로운 목표 각도 수신 - 이때만 target_q 업데이트"""
        if not self.initial_posture_captured or self.emergency_stop:
            return
            
        matched_joints = 0
        angle_changes = {}
        
        for joint_name, g1_idx in ROS_TO_G1_INDEX.items():
            if joint_name in msg.name:
                idx_in_msg = msg.name.index(joint_name)
                new_angle = float(msg.position[idx_in_msg])
                old_angle = self.target_q[g1_idx]
                
                if abs(new_angle) > np.pi:
                    self.get_logger().warn(f"⚠️ {joint_name}: {new_angle:.3f}rad 범위 초과")
                    continue
                
                # 변화량 추적 (디버깅용)
                diff = abs(new_angle - old_angle)
                if diff > 0.001:  # 0.001 라디안 이상 변화만 기록
                    short_name = joint_name.replace('right_', '').replace('_joint', '')
                    angle_changes[short_name] = f"{old_angle:.4f}→{new_angle:.4f}"
                
                # ⭐️ VR 명령으로 목표 각도 업데이트
                self.target_q[g1_idx] = new_angle
                matched_joints += 1
        
        if matched_joints > 0:
            self.last_vr_time = time.time()
            if not self.vr_active:
                self.vr_active = True
                self.get_logger().info("🎮 VR 제어 시작! 로봇이 부드럽게 따라갑니다")
        
        # 실시간 상태 로그
        now = time.time()
        if now - self.last_log > 2.0:
            if angle_changes:
                self.get_logger().info(f"📡 VR 명령: {matched_joints}개 관절 | 🎯 변화: {angle_changes}")
            elif matched_joints > 0:
                self.get_logger().info(f"📡 VR 명령: {matched_joints}개 관절 (미세 조정)")
            else:
                self.get_logger().info("📡 VR 연결 유지 중")
            self.last_log = now

    def LowCmdWrite(self):
        """⭐️ 500Hz 메인 제어 루프 - 단순화된 구조"""
        if not self.initial_posture_captured or self.low_state is None:
            return

        self.low_cmd.mode_pr = Mode.PR
        self.low_cmd.mode_machine = self.mode_machine_

        if self.emergency_stop:
            self._send_hold_position()
        else:
            # ⭐️ 항상 증분 제어만 사용 (초기화 단계 완전 제거)
            self._smooth_teleop()

        # CRC 계산 후 전송
        crc_ok = set_low_cmd_crc(self.low_cmd)
        self.lowcmd_publisher_.Write(self.low_cmd)
        self.cmd_count += 1

        # 주기적 상태 로그
        if self.cmd_count % 2500 == 0:  # 5초마다
            crc_status = "✅" if crc_ok else "⚠️"
            vr_status = "활성" if self.vr_active else "대기"
            self.get_logger().info(
                f'{crc_status} DDS 전송 {self.cmd_count}회 | '
                f'VR: {vr_status} | CRC: 0x{self.low_cmd.crc:08X}'
            )

    def _smooth_teleop(self):
        """
        ⭐️ 핵심 로직: VR 목표(target_q)를 향해 아주 조금씩 이동
        - VR 명령이 없으면 target_q == current_smooth_q 이므로 움직임 없음
        - VR 명령이 들어오면 MAX_DELTA 속도로 부드럽게 이동
        """
        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0

            if i in RIGHT_ARM_INDICES:
                # ⭐️ 오른팔: VR 목표를 향해 증분 이동
                target = self.target_q[i]
                current = self.current_smooth_q[i]
                diff = target - current
                
                # 매우 작은 변화량으로 부드럽게 이동
                if abs(diff) > self.MAX_DELTA:
                    self.current_smooth_q[i] = current + np.sign(diff) * self.MAX_DELTA
                else:
                    self.current_smooth_q[i] = target

                # 부드러운 게인 적용
                self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                self.low_cmd.motor_cmd[i].kp = Kp[i] * self.SOFT_KP_RATIO
                self.low_cmd.motor_cmd[i].kd = Kd[i] * self.SOFT_KD_RATIO
            else:
                # ⭐️ 나머지 관절: 초기 위치 완전 유지 (절대 건드리지 않음)
                self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
                self.low_cmd.motor_cmd[i].kp = Kp[i]
                self.low_cmd.motor_cmd[i].kd = Kd[i]

    def _send_hold_position(self):
        """비상 정지: 현재 부드러운 각도 유지"""
        for i in range(G1_NUM_MOTOR):
            self.low_cmd.motor_cmd[i].mode = 1
            self.low_cmd.motor_cmd[i].q = self.current_smooth_q[i]
            self.low_cmd.motor_cmd[i].dq = 0.0
            self.low_cmd.motor_cmd[i].tau = 0.0
            self.low_cmd.motor_cmd[i].kp = Kp[i]
            self.low_cmd.motor_cmd[i].kd = Kd[i]

    def keyboard_monitor(self):
        """키보드 제어"""
        print("\n⌨️  키보드 제어:")
        print("   s + Enter: 비상 정지 (현재 자세 유지)")
        print("   r + Enter: 제어 재개")
        print("   + + Enter: 속도 증가 (x1.5)")
        print("   - + Enter: 속도 감소 (÷1.5)")
        print(f"   현재 MAX_DELTA: {self.MAX_DELTA:.5f} rad/tick\n")
        
        try:
            while True:
                key = input().strip().lower()
                if key in ['s', 'stop']:
                    self.emergency_stop = True
                    self.get_logger().warn('🛑 비상 정지! (현재 자세 유지)')
                elif key in ['r', 'resume']:
                    self.emergency_stop = False
                    self.get_logger().info('▶️ 제어 재개')
                elif key == '+':
                    self.MAX_DELTA = min(self.MAX_DELTA * 1.5, 0.01)
                    self.get_logger().info(f'⚡ 속도 증가: MAX_DELTA={self.MAX_DELTA:.5f}')
                elif key == '-':
                    self.MAX_DELTA = max(self.MAX_DELTA / 1.5, 0.0001)
                    self.get_logger().info(f'🐢 속도 감소: MAX_DELTA={self.MAX_DELTA:.5f}')
        except (EOFError, KeyboardInterrupt):
            pass


def main():
    print("=" * 70)
    print("🤖 Meta Quest3 → G1 순수 텔레오퍼레이션 브릿지")
    print("⭐ 특징: 브릿지만 실행해도 로봇이 스스로 움직이지 않음")
    print("⚠️  주의: 로봇 주변에 장애물이 없는지 확인하세요!")
    print("=" * 70)
    
    if len(sys.argv) < 2:
        print("❌ 사용법: python3 g1_bridge_node.py enx80691ad9f138")
        sys.exit(1)

    iface = sys.argv[1]
    print(f"🌐 네트워크 인터페이스: {iface}")

    # DDS 초기화
    ChannelFactoryInitialize(0, iface)
    time.sleep(0.5)

    # ROS2 노드 시작
    rclpy.init()
    node = G1TeleopBridge()
    
    try:
        node.Init()
        node.Start()
        rclpy.spin(node)
    except KeyboardInterrupt:
        print("\n🛑 브릿지 노드 종료 중...")
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
