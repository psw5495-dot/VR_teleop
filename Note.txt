def parse_frame_from_json(data: dict) -> VRFrame:
    return VRFrame(
        hand=data.get('hand', 'right'),
        x=float(data.get('x', 0)),
        y=float(data.get('y', 0)),
        z=float(data.get('z', 0)),
        qx=float(data.get('qx', 0)),
        qy=float(data.get('qy', 0)),
        qz=float(data.get('qz', 0)),
        qw=float(data.get('qw', 1)),
        # ⭐️ 서버 수신 시간으로 타임스탬프 통일 (Quest3 클럭 불일치 방지)
        timestamp=time.time()
    )
