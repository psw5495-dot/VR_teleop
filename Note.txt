#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped
from aiohttp import web
import json
import ssl
import threading
import argparse
import time

# ROS 2 노드
class VRTeleopNode(Node):
    def __init__(self):
        super().__init__('vr_teleop_safe')
        self.right_pub = self.create_publisher(PoseStamped, '/quest3/right_hand_pose', 10)
        self.msg_count = 0
        self.get_logger().info("✅ VR Teleop Safe 노드 시작")
        
    def publish_pose(self, x, y, z, qx, qy, qz, qw):
        msg = PoseStamped()
        msg.header.stamp = self.get_clock().now().to_msg()
        msg.header.frame_id = 'quest3_world'
        msg.pose.position.x, msg.pose.position.y, msg.pose.position.z = float(x), float(y), float(z)
        msg.pose.orientation.x, msg.pose.orientation.y = float(qx), float(qy)
        msg.pose.orientation.z, msg.pose.orientation.w = float(qz), float(qw)
        self.right_pub.publish(msg)
        self.msg_count += 1

vr_node = None

def ros_spin():
    rclpy.init()
    global vr_node
    vr_node = VRTeleopNode()
    try:
        rclpy.spin(vr_node)
    except:
        pass
    finally:
        vr_node.destroy_node()
        rclpy.shutdown()

# WebSocket
async def ws_handler(request):
    ws = web.WebSocketResponse()
    await ws.prepare(request)
    try:
        async for msg in ws:
            if msg.type == web.WSMsgType.TEXT:
                data = json.loads(msg.data)
                if vr_node and 'hand' in data and data['hand'] == 'right':
                    vr_node.publish_pose(
                        data['x'], data['y'], data['z'],
                        data['qx'], data['qy'], data['qz'], data['qw']
                    )
    except Exception as e:
        print(f"WS Error: {e}")
    return ws

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--port', type=int, default=8443)
    parser.add_argument('--ssl-cert', required=True)
    parser.add_argument('--ssl-key', required=True)
    args = parser.parse_args()

    threading.Thread(target=ros_spin, daemon=True).start()
    time.sleep(1)

    app = web.Application()
    app.router.add_get('/', index_handler)
    app.router.add_get('/ws', ws_handler)

    ssl_context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ssl_context.load_cert_chain(args.ssl_cert, args.ssl_key)

    print(f"🚀 Safe VR 서버: https://0.0.0.0:{args.port}")
    web.run_app(app, port=args.port, ssl_context=ssl_context, print=None)

if __name__ == '__main__':
    main()
