#!/usr/bin/env python3
# ~/TeleVision/g1_bridge_node.py
"""
ROS2 → G1 DDS 브릿지 노드 (변수 스코프 및 API 호환성 완전 수정)
"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState
import time
import subprocess

# ── ⭐️ 전역 변수 사전 정의 (스코프 문제 해결) ─────────────────────
SDK_AVAILABLE = False
LowCmd_ = None
ChannelPublisher = None
init_function = None

# ── Unitree SDK2 임포트 (여러 방법 순차 시도) ────────────────────
print("🔍 Unitree SDK2 로드 시도 중...")

# 1단계: LowCmd_ 메시지 타입 찾기
try:
    from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_
    print("✅ LowCmd_ 로드: unitree_hg 경로")
except ImportError:
    try:
        from unitree_sdk2py.idl.unitree_go.msg.dds_ import LowCmd_
        print("✅ LowCmd_ 로드: unitree_go 경로")
    except ImportError:
        try:
            from unitree_sdk2py.idl.default import unitree_hg_msg_dds__LowCmd_ as LowCmd_
            print("✅ LowCmd_ 로드: default 경로")
        except ImportError:
            print("❌ LowCmd_ 로드 실패")
            LowCmd_ = None

# 2단계: 채널 퍼블리셔 임포트
if LowCmd_ is not None:
    try:
        from unitree_sdk2py.core.channel import ChannelPublisher
        print("✅ ChannelPublisher 로드 성공")
    except ImportError:
        print("❌ ChannelPublisher 로드 실패")
        ChannelPublisher = None

# 3단계: DDS 초기화 함수 찾기 (여러 방법 시도)
if LowCmd_ is not None and ChannelPublisher is not None:
    # 방법 1: ChannelFactoryInitialize (가장 일반적)
    try:
        from unitree_sdk2py.core.channel import ChannelFactoryInitialize
        init_function = ChannelFactoryInitialize
        print("✅ DDS 초기화: ChannelFactoryInitialize")
        SDK_AVAILABLE = True
    except ImportError:
        pass
    
    # 방법 2: InitChannels (대안)
    if not SDK_AVAILABLE:
        try:
            from unitree_sdk2py.core.channel import InitChannels
            init_function = InitChannels
            print("✅ DDS 초기화: InitChannels")
            SDK_AVAILABLE = True
        except ImportError:
            pass
    
    # 방법 3: ChannelFactory 클래스 (최후 수단)
    if not SDK_AVAILABLE:
        try:
            from unitree_sdk2py.core.channel import ChannelFactory
            # 클래스 인스턴스 생성 방식으로 시도
            def init_via_class(domain_id, interface):
                ChannelFactory(domain_id, interface)
            init_function = init_via_class
            print("✅ DDS 초기화: ChannelFactory 클래스")
            SDK_AVAILABLE = True
        except ImportError:
            pass

# 최종 상태 출력
if SDK_AVAILABLE:
    print(f"🎉 Unitree SDK2 완전 로드 성공!")
else:
    print("⚠️ Unitree SDK2 로드 실패 → 시뮬레이션 모드로 동작")

# ── G1 모터 인덱스 매핑 ───────────────────────────────────────
G1_RIGHT_ARM_MOTOR_IDX = {
    'right_shoulder_pitch_joint': 15,
    'right_shoulder_roll_joint' : 16,
    'right_shoulder_yaw_joint'  : 17,
    'right_elbow_pitch_joint'   : 18,
    'right_wrist_roll_joint'    : 19,
    'right_wrist_pitch_joint'   : 20,
    'right_wrist_yaw_joint'     : 21,
}


class G1BridgeNode(Node):
    def __init__(self):
        super().__init__('g1_bridge_node')
        
        # SDK_AVAILABLE은 이제 항상 정의되어 있음
        if SDK_AVAILABLE:
            try:
                self._init_dds()
            except Exception as e:
                self.get_logger().error(f'❌ DDS 초기화 실패: {e}')
                self.get_logger().warn('🔵 시뮬레이션 모드로 전환')
                self.cmd_pub = None
        else:
            self.get_logger().warn('⚠️ SDK 없음 - 시뮬레이션 모드')
            self.cmd_pub = None
        
        # ROS2 구독
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', self.joint_callback, 10)
        
        # 제어 파라미터
        self.KP = 40.0
        self.KD = 1.0
        self.MAX_TORQUE = 10.0
        
        self.cmd_count = 0
        self.last_log = time.time()
        
        self.get_logger().info('✅ G1 브릿지 노드 시작')
        self.get_logger().info('📡 /g1/arm/joint_commands 구독 중...')

    def _init_dds(self):
        """DDS 채널 초기화"""
        iface = self._detect_network_interface()
        self.get_logger().info(f'🌐 DDS 인터페이스: {iface}')
        
        # 감지된 초기화 함수 사용
        init_function(0, iface)
        self.get_logger().info('✅ DDS 초기화 완료')
        
        # 퍼블리셔 생성
        self.cmd_pub = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.cmd_pub.Init()
        
        # LowCmd 메시지 초기화
        self.low_cmd = LowCmd_()
        try:
            # motor_cmd 배열 크기 자동 감지
            motor_count = len(self.low_cmd.motor_cmd)
            self.get_logger().info(f'📊 G1 모터 수: {motor_count}개')
            
            for i in range(motor_count):
                self.low_cmd.motor_cmd[i].q   = 0.0
                self.low_cmd.motor_cmd[i].dq  = 0.0
                self.low_cmd.motor_cmd[i].kp  = 0.0
                self.low_cmd.motor_cmd[i].kd  = 0.5
                self.low_cmd.motor_cmd[i].tau = 0.0
                
        except Exception as e:
            self.get_logger().warn(f'⚠️ LowCmd 초기화 부분 실패: {e}')
            # 기본값으로 35개 모터 가정
            for i in range(35):
                self.low_cmd.motor_cmd[i].q   = 0.0
                self.low_cmd.motor_cmd[i].dq  = 0.0
                self.low_cmd.motor_cmd[i].kp  = 0.0
                self.low_cmd.motor_cmd[i].kd  = 0.5
                self.low_cmd.motor_cmd[i].tau = 0.0

    def _detect_network_interface(self):
        """네트워크 인터페이스 자동 감지"""
        try:
            result = subprocess.run(
                ['ip', 'route', 'get', '8.8.8.8'],
                capture_output=True, text=True, timeout=3
            )
            tokens = result.stdout.split()
            for i, token in enumerate(tokens):
                if token == 'dev' and i + 1 < len(tokens):
                    return tokens[i + 1]
                elif token.startswith(('eth', 'enp', 'eno', 'ens')):
                    return token
        except Exception:
            pass
        return "AUTO"

    def joint_callback(self, msg):
        """ROS2 JointState → G1 DDS 명령 변환"""
        try:
            if not SDK_AVAILABLE or self.cmd_pub is None:
                self._log_simulation(msg)
                return
            
            # 관절 매핑
            for joint_name, motor_idx in G1_RIGHT_ARM_MOTOR_IDX.items():
                if joint_name not in msg.name:
                    continue
                
                j_idx = msg.name.index(joint_name)
                if j_idx >= len(msg.position):
                    continue
                
                target_q = float(msg.position[j_idx])
                
                # 안전 범위 체크 (±π 라디안)
                if abs(target_q) > 3.14159:
                    self.get_logger().warn(f'⚠️ {joint_name}: 각도 {target_q:.3f} 범위 초과')
                    continue
                
                # LowCmd 설정
                self.low_cmd.motor_cmd[motor_idx].q   = target_q
                self.low_cmd.motor_cmd[motor_idx].dq  = 0.0
                self.low_cmd.motor_cmd[motor_idx].kp  = self.KP
                self.low_cmd.motor_cmd[motor_idx].kd  = self.KD
                self.low_cmd.motor_cmd[motor_idx].tau = 0.0
            
            # DDS 전송
            self.cmd_pub.Write(self.low_cmd)
            self.cmd_count += 1
            
            # 주기적 로그
            now = time.time()
            if now - self.last_log > 5.0:
                self.get_logger().info(
                    f'✅ DDS 전송 {self.cmd_count}회 | '
                    f'관절: {len(msg.position)}개'
                )
                self.last_log = now
                
        except Exception as e:
            self.get_logger().error(f'❌ 브릿지 에러: {e}')

    def _log_simulation(self, msg):
        """시뮬레이션 모드 로그"""
        now = time.time()
        if now - self.last_log > 3.0:
            if len(msg.position) > 0:
                pos_str = ', '.join(f'{p:.3f}' for p in msg.position[:3])
                self.get_logger().info(f'🔵 [시뮬] 관절 명령: [{pos_str}...]')
            else:
                self.get_logger().info('🔵 [시뮬] 빈 관절 명령 수신')
            self.last_log = now


def main():
    rclpy.init()
    node = G1BridgeNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        node.get_logger().info('🛑 브릿지 노드 종료')
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
