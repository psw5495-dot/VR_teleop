#!/usr/bin/env python3
# ~/g1_teleop/g1_bridge_node.py
"""
ROS2 → G1 DDS 브릿지 노드
Ubuntu PC에서 G1으로 직접 DDS 명령 전송
"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import JointState
import numpy as np
import time
import subprocess

# ── Unitree SDK2 임포트 ───────────────────────────────────────
try:
    from unitree_sdk2py.core.channel import ChannelFactory, ChannelPublisher
    from unitree_sdk2py.idl.unitree_hg.msg.dds_ import LowCmd_
    SDK_AVAILABLE = True
    print("✅ Unitree SDK2 로드 성공")
except ImportError as e:
    SDK_AVAILABLE = False
    print(f"⚠️ Unitree SDK2 없음: {e}")

# ── G1 모터 인덱스 매핑 ───────────────────────────────────────
G1_RIGHT_ARM_MOTOR_IDX = {
    'right_shoulder_pitch_joint': 15,
    'right_shoulder_roll_joint' : 16,
    'right_shoulder_yaw_joint'  : 17,
    'right_elbow_pitch_joint'   : 18,
    'right_wrist_roll_joint'    : 19,
    'right_wrist_pitch_joint'   : 20,
    'right_wrist_yaw_joint'     : 21,
}


class G1BridgeNode(Node):
    def __init__(self):
        super().__init__('g1_bridge_node')
        
        # DDS 초기화
        if SDK_AVAILABLE:
            self._init_dds()
        else:
            self.get_logger().warn('⚠️ DDS 없이 실행 (시뮬레이션 모드)')
            self.cmd_pub = None
        
        # ROS2 구독
        self.subscription = self.create_subscription(
            JointState, '/g1/arm/joint_commands', self.joint_callback, 10)
        
        # 안전 파라미터
        self.KP = 40.0
        self.KD = 1.0
        self.MAX_TORQUE = 10.0
        
        self.cmd_count = 0
        self.last_log = time.time()
        
        self.get_logger().info('✅ G1 브릿지 노드 시작')
        self.get_logger().info('📡 /g1/arm/joint_commands 구독 중...')

    def _init_dds(self):
        """DDS 채널 초기화"""
        # 네트워크 인터페이스 자동 감지
        iface = self._detect_network_interface()
        self.get_logger().info(f'🌐 DDS 인터페이스: {iface}')
        
        ChannelFactory.Instance().Init(0, iface)
        
        # LowCmd 퍼블리셔
        self.cmd_pub = ChannelPublisher("rt/lowcmd", LowCmd_)
        self.cmd_pub.Init()
        
        # LowCmd 메시지 초기화
        self.low_cmd = LowCmd_()
        for i in range(35):  # G1 전체 모터 초기화
            self.low_cmd.motor_cmd[i].q   = 0.0
            self.low_cmd.motor_cmd[i].dq  = 0.0
            self.low_cmd.motor_cmd[i].kp  = 0.0
            self.low_cmd.motor_cmd[i].kd  = 0.5
            self.low_cmd.motor_cmd[i].tau = 0.0

    def _detect_network_interface(self):
        """G1과 연결된 네트워크 인터페이스 자동 감지"""
        try:
            result = subprocess.run(
                ['ip', 'route', 'get', '8.8.8.8'],
                capture_output=True, text=True
            )
            for token in result.stdout.split():
                # 이더넷 인터페이스 우선 선택
                if token.startswith(('eth', 'enp', 'eno', 'ens')):
                    return token
        except Exception:
            pass
        return "AUTO"

    def joint_callback(self, msg):
        """ROS2 JointState → G1 DDS 명령 변환"""
        try:
            if not SDK_AVAILABLE or self.cmd_pub is None:
                self._log_simulation(msg)
                return
            
            # 관절 이름 → 모터 인덱스 매핑
            for joint_name, motor_idx in G1_RIGHT_ARM_MOTOR_IDX.items():
                if joint_name not in msg.name:
                    continue
                
                j_idx = msg.name.index(joint_name)
                if j_idx >= len(msg.position):
                    continue
                
                target_q = float(msg.position[j_idx])
                
                # 모터 명령 설정 (위치 제어)
                self.low_cmd.motor_cmd[motor_idx].q   = target_q
                self.low_cmd.motor_cmd[motor_idx].dq  = 0.0
                self.low_cmd.motor_cmd[motor_idx].kp  = self.KP
                self.low_cmd.motor_cmd[motor_idx].kd  = self.KD
                self.low_cmd.motor_cmd[motor_idx].tau = 0.0
            
            # G1으로 DDS 전송
            self.cmd_pub.Write(self.low_cmd)
            self.cmd_count += 1
            
            # 5초마다 상태 로그
            now = time.time()
            if now - self.last_log > 5.0:
                self.get_logger().info(
                    f'✅ DDS 전송 {self.cmd_count}회 | 관절: {len(msg.position)}개')
                self.last_log = now
                
        except Exception as e:
            self.get_logger().error(f'❌ 브릿지 에러: {e}')

    def _log_simulation(self, msg):
        """시뮬레이션 모드 로그"""
        now = time.time()
        if now - self.last_log > 3.0:
            pos_str = ', '.join(f'{p:.3f}' for p in msg.position[:3])
            self.get_logger().info(f'🔵 [시뮬] 관절 명령: [{pos_str}...]')
            self.last_log = now


def main():
    rclpy.init()
    node = G1BridgeNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        node.get_logger().info('🛑 브릿지 노드 종료')
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
